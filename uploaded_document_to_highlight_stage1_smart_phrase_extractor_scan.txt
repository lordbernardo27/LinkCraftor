   378: def canonical_phrase(text: str) -> str:
   379:     s = (text or "").strip().lower()
   380:     s = s.replace("\u2019", "'").replace("\u2018", "'")
   381:     s = s.replace("\u201c", '"').replace("\u201d", '"')
   382:     s = re.sub(r"^\s*(?:\d+[.)]\s+|[\u2022\u2013\u2014\-]\s+)", "", s)
   383:     s = re.sub(r"^[\"'(\[{]+|[\"')\]}:;,.!?]+$", "", s)
   384:     s = re.sub(r"\s+", " ", s).strip()
   385:     return s

   454: def _heuristic_tag(word: str) -> str:
   455:     w = word.lower()
   456:     if w in FUNCTION_WORDS:
   457:         if w in DETERMINERS:
   458:             return "DET"
   459:         if w in PREPOSITIONS:
   460:             return "ADP"
   461:         if w in CONJUNCTIONS:
   462:             return "CONJ"
   463:         if w in PRONOUNS:
   464:             return "PRON"
   465:         if w in AUX_VERBS:
   466:             return "AUX"
   467:         return "ADV"
   468:     if w.isdigit():
   469:         return "NUM"
   470:     if w in ADJECTIVAL_LY:
   471:         return "ADJ"
   472:     if w in _ING_ED_NOUNS:
   473:         return "NOUN"
   474:     # inflected verb forms (builds, boosts, tracks) are strong clause signals;
   475:     # the context retag pass can still demote a clause-final one to a plural noun.
   476:     if _is_inflected_verb(w):
   477:         return "VERB"
   478:     if w in GERUND_FORMS:
   479:         return "NOUN"        # gerunds usually nominal/modifier in NP context
   480:     # known noun heads / modifiers and noun morphology win over a *bare* lemma
   481:     # that merely happens to also be a verb (rate, score, support, schedule).
   482:     if w in UNIVERSAL_HEADS or w in UNIVERSAL_MODIFIERS or w in GENERIC_WEAK_HEADS:
   483:         return "NOUN"
   484:     if w.endswith(_NOUN_SUFFIXES):
   485:         return "NOUN"
   486:     # remaining bare verb lemmas (make, reduce, build, drive...) are clause verbs
   487:     if w in FINITE_VERB_FORMS or w in ACTION_VERBS:
   488:         return "VERB"
   489:     if w.endswith("ly") and len(w) > 4:
   490:         return "ADV"
   491:     if w.endswith(_VERB_SUFFIXES):
   492:         return "VERB"
   493:     # default: treat unknown content words as nouns (safer for NP chunking)
   494:     return "NOUN"

   497: def pos_tag(tokens: List[str]) -> List[Tuple[str, str]]:
   498:     """Return [(token, coarse_tag), ...] using the best available tagger."""
   499:     if not tokens:
   500:         return []
   501:     if _have_nltk():
   502:         try:
   503:             import nltk  # type: ignore
   504:             raw = nltk.pos_tag(tokens)
   505:             return [(w, _coarsen_ptb(t)) for w, t in raw]
   506:         except Exception:
   507:             pass
   508:     return [(t, _heuristic_tag(t)) for t in tokens]

   511: def _coarsen_ptb(tag: str) -> str:
   512:     if tag.startswith("NN"):
   513:         return "NOUN"
   514:     if tag.startswith("VB"):
   515:         return "VERB"
   516:     if tag.startswith("JJ"):
   517:         return "ADJ"
   518:     if tag.startswith("RB"):
   519:         return "ADV"
   520:     if tag in {"DT", "PDT", "WDT"}:
   521:         return "DET"
   522:     if tag == "IN":
   523:         return "ADP"
   524:     if tag == "CC":
   525:         return "CONJ"
   526:     if tag in {"PRP", "PRP$", "WP", "WP$"}:
   527:         return "PRON"
   528:     if tag == "CD":
   529:         return "NUM"
   530:     if tag == "MD":
   531:         return "AUX"
   532:     return "X"

   539: def _trim_boundaries(tokens: Sequence[str]) -> List[str]:
   540:     """Strip leading/trailing function words so the phrase begins and ends on a
   541:     content word. Adjectival ``-ly`` modifiers are preserved at the front.
   542: 
   543:     Only *inflected* finite verb forms (boosts, builds, tracks) are stripped
   544:     from the front; bare lemmas that double as nouns (search, lean, feature)
   545:     are kept, since here they head/modify a noun phrase."""
   546:     toks = list(tokens)
   547:     while toks and (
   548:         toks[0] in LEADING_BAN
   549:         or (_is_inflected_verb(toks[0]) and toks[0] not in ACTION_VERBS)
   550:     ):
   551:         toks.pop(0)
   552:     while toks and (
   553:         toks[-1] in TRAILING_BAN
   554:         or toks[-1] in AUX_VERBS
   555:         or toks[-1] in PARTICLES_DEGREE
   556:         or _is_adverb_ly(toks[-1])
   557:     ):
   558:         toks.pop()
   559:     return toks

   562: def _head_is_noun_like(token: str, tag: Optional[str]) -> bool:
   563:     """Decide whether ``token`` can be a phrase head. When a tagger supplies a
   564:     tag we trust it (this is what makes heads niche-agnostic); otherwise we use
   565:     morphology + the closed verb class."""
   566:     if token in COMPARATIVES or _is_adverb_ly(token):
   567:         return False
   568:     if tag is not None:
   569:         if tag in {"NOUN", "NUM"}:
   570:             return True
   571:         if tag == "ADJ":
   572:             return False  # adjective tail = hanging ("very large")
   573:         if tag == "VERB":
   574:             # gerund / participle used nominally is OK; finite verb is not.
   575:             return token in _ING_ED_NOUNS or token in GERUND_FORMS
   576:         return False
   577:     # ---- no tagger: morphology-based fallback ----
   578:     if token in _ING_ED_NOUNS or token in GERUND_FORMS:
   579:         return True
   580:     if token in GENERIC_WEAK_HEADS or token in UNIVERSAL_HEADS:
   581:         return True
   582:     if token in FINITE_VERB_FORMS:
   583:         return False
   584:     if token in (AUX_VERBS | PREPOSITIONS | CONJUNCTIONS | PRONOUNS | DETERMINERS):
   585:         return False
   586:     if token.endswith("ed") and token not in _ING_ED_NOUNS:
   587:         return False
   588:     return True

   591: def _is_finite_verb(token: str, tag: Optional[str]) -> bool:
   592:     """Is this token a finite/clause verb (a clause signal), as opposed to a
   593:     nominal gerund/participle modifier? Tags have already been cleaned by the
   594:     retag pass, so when a tag is present we trust it."""
   595:     if _is_gerund(token) or token in _ING_ED_NOUNS or token in GERUND_FORMS:
   596:         return False
   597:     if tag == "VERB":
   598:         return True
   599:     if tag in {"NOUN", "NUM", "ADJ"}:
   600:         return False                         # trust the (retagged) tagger
   601:     # no tag available: fall back to the closed verb class, inflected forms only
   602:     return _is_inflected_verb(token) or (token in ALL_VERB_LEMMAS
   603:                                          and token in FINITE_VERB_FORMS
   604:                                          and token not in _ING_ED_NOUNS)

   607: def _is_complete_phrase(tokens: Sequence[str],
   608:                         tags: Optional[Sequence[str]] = None) -> bool:
   609:     """A phrase is complete iff it is a clean noun-phrase constituent.
   610: 
   611:     Tag-aware: when ``tags`` are provided they drive the head and interior-verb
   612:     decisions, so any niche's head noun is accepted. ``tags`` must align 1:1
   613:     with ``tokens``.
   614:     """
   615:     toks = list(tokens)
   616:     if not (2 <= len(toks) <= 6):
   617:         return False
   618:     tg: List[Optional[str]] = list(tags) if tags is not None else [None] * len(toks)
   619:     if len(tg) != len(toks):
   620:         tg = [None] * len(toks)
   621: 
   622:     if toks[0] in LEADING_BAN or toks[-1] in TRAILING_BAN:
   623:         return False
   624:     if not _head_is_noun_like(toks[-1], tg[-1]):
   625:         return False
   626:     if any(t in AUX_VERBS for t in toks):
   627:         return False
   628:     if any(t in PRONOUNS for t in toks):
   629:         return False
   630:     # leading finite verb (not an action verb) => glued clause ("boosts metabolism")
   631:     if _is_finite_verb(toks[0], tg[0]) and toks[0] not in ACTION_VERBS:
   632:         return False
   633:     # any interior finite verb => glued clause
   634:     for t, g in zip(toks[1:-1], tg[1:-1]):
   635:         if _is_finite_verb(t, g):
   636:             return False
   637:     if any(t in STRIP_INTERIOR for t in toks[1:-1]):
   638:         return False
   639:     # interior conjunctions / non-'of' prepositions => fragment
   640:     for t in toks[1:-1]:
   641:         if t in CONJUNCTIONS or (t in PREPOSITIONS and t != "of"):
   642:             return False
   643:     content = [t for t in toks if t not in FUNCTION_WORDS]
   644:     if len(content) < 2:
   645:         return False
   646:     if all(t in GENERIC_WEAK_HEADS or t in FUNCTION_WORDS for t in toks):
   647:         return False
   648:     return True

   651: def _is_complete_intent(tokens: Sequence[str],
   652:                         tags: Optional[Sequence[str]] = None) -> bool:
   653:     """Validate a verb-bearing intent phrase ("how to calculate fertile window").
   654: 
   655:     Intent phrases legitimately contain an interior verb, so they use their own
   656:     rule: recognised intent prefix, 3..7 tokens, ends on a noun-like head, no
   657:     dangling boundary, no pronouns."""
   658:     toks = list(tokens)
   659:     if not (3 <= len(toks) <= 7):
   660:         return False
   661:     if any(t in PRONOUNS for t in toks):
   662:         return False
   663:     if toks[-1] in TRAILING_BAN or toks[-1] in AUX_VERBS:
   664:         return False
   665:     joined = " ".join(toks)
   666:     if not any(joined.startswith(p) for p in INTENT_STARTS):
   667:         return False
   668:     tg = list(tags) if tags is not None and len(tags) == len(toks) else [None] * len(toks)
   669:     if not _head_is_noun_like(toks[-1], tg[-1]):
   670:         return False
   671:     return True

   707: def _trim_with_tags(tokens: List[str], tags: List[str]) -> Tuple[List[str], List[str]]:
   708:     toks, tg = list(tokens), list(tags)
   709:     while toks and (
   710:         toks[0] in LEADING_BAN
   711:         or (_confirmed_finite(toks[0], tg[0]) and toks[0] not in ACTION_VERBS)
   712:     ):
   713:         toks.pop(0)
   714:         tg.pop(0)
   715:     while toks and (
   716:         toks[-1] in TRAILING_BAN or toks[-1] in AUX_VERBS
   717:         or toks[-1] in PARTICLES_DEGREE or _is_adverb_ly(toks[-1])
   718:     ):
   719:         toks.pop()
   720:         tg.pop()

   728: def _is_gerund(w: str) -> bool:
   729:     """True for -ing forms that act as nominal/participial NP modifiers."""
   730:     return len(w) > 4 and w.endswith("ing") and w not in AUX_VERBS

   733: def _confirmed_finite(w: str, t: Optional[str]) -> bool:
   734:     """A *confirmed* finite verb: an inflected known verb form (strong signal),
   735:     or a base verb the tagger also calls a VERB. Base forms that are also common
   736:     nouns (feature, search, rate, ...) are NOT confirmed unless the tagger agrees,
   737:     so we never shatter a noun compound on an ambiguous word."""
   738:     if _is_gerund(w) or w in _ING_ED_NOUNS:
   739:         return False
   740:     if w in FINITE_VERB_FORMS:
   741:         if w in ALL_VERB_LEMMAS:        # base form -> need tagger agreement
   742:             return t == "VERB"
   743:         return True                      # inflected -s/-es/-ed/-ies -> strong verb
   744:     return False

   747: def _np_eligible(w: str, t: Optional[str]) -> bool:
   748:     """May this token sit *inside* a noun phrase run? Tags are already retagged,
   749:     so VERB means a real clause verb here."""
   750:     if w == "to":
   751:         return False
   752:     if t in {"DET", "CONJ", "PRON", "AUX"}:
   753:         return False
   754:     if t == "ADP" and w != "of":
   755:         return False
   756:     if w in DETERMINERS or w in CONJUNCTIONS or w in PRONOUNS or w in AUX_VERBS:
   757:         return False
   758:     if w in PREPOSITIONS and w != "of":
   759:         return False
   760:     if _is_gerund(w) or w in _ING_ED_NOUNS:
   761:         return True
   762:     if t == "VERB":
   763:         return False
   764:     return True

   767: def _is_inflected_verb(w: str) -> bool:
   768:     """A known verb in an inflected (-s/-es/-ed/-ies) form, i.e. NOT a bare
   769:     lemma. These are the forms NLTK most often mislabels as NNS in compounds."""
   770:     return (w in FINITE_VERB_FORMS and w not in ALL_VERB_LEMMAS
   771:             and not _is_gerund(w) and w not in _ING_ED_NOUNS)

   774: _OBJECT_STARTERS = {"NOUN", "ADJ", "NUM", "DET", "PRON"}

   776: 
   777: def _nominalize_tags(tags: List[Tuple[str, str]]) -> List[Tuple[str, str]]:
   778:     """Context-aware retag that cleans up the two systematic tagger errors that
   779:     fragment noun compounds, then lets everything downstream simply trust tags:
   780: 
   781:     1. Inflected verb forms (builds, tracks, boosts, rates) are decided by
   782:        right context: a finite verb is followed by an object/predicate; the same
   783:        form at clause end / before a conjunction or preposition is a plural noun.
   784:     2. A bare verb lemma the tagger called VERB but which sits inside an object
   785:        NP after the real clause verb (and not after to/of/det) is a compound
   786:        noun modifier ('tracks feature adoption')."""
   787:     out = list(tags)
   788:     n = len(out)
   789:     block_left = ({"to"} | PREPOSITIONS | CONJUNCTIONS | AUX_VERBS
   790:                   | DETERMINERS | PRONOUNS)
   791:     seen_verb = False
   792:     for i, (w, t) in enumerate(out):
   793:         if w in CONJUNCTIONS or w == "to":
   794:             seen_verb = False
   795:         rt = out[i + 1] if i + 1 < n else None
   796: 
   797:         # (1) resolve inflected known-verb forms by right context
   798:         if _is_inflected_verb(w):
   799:             obj_start = (
   800:                 rt is not None and rt[0] not in CONJUNCTIONS and rt[0] != "to"
   801:                 and (rt[1] in _OBJECT_STARTERS
   802:                      or (rt[1] == "VERB" and rt[0] in ALL_VERB_LEMMAS
   803:                          and not _is_gerund(rt[0])))
   804:             )
   805:             out[i] = (w, "VERB" if obj_start else "NOUN")
   806:             if obj_start:
   807:                 seen_verb = True
   808:             continue
   809: 
   810:         # (2) demote a bare verb lemma that is really a compound modifier
   811:         is_base = (w in ALL_VERB_LEMMAS and w not in _ING_ED_NOUNS
   812:                    and not _is_gerund(w))
   813:         if is_base and t == "VERB" and seen_verb:
   814:             lt = out[i - 1] if i > 0 else None
   815:             right_noun = rt is not None and _head_is_noun_like(rt[0], rt[1])
   816:             left_ok = lt is not None and lt[0] not in block_left
   817:             if right_noun and left_ok:
   818:                 out[i] = (w, "NOUN")
   819:                 continue
   820: 
   821:         # a real present-tense finite verb marks the clause verb as seen;
   822:         # leading -ed participles (automated, integrated) must NOT count.
   823:         if (t == "VERB" and not _is_gerund(w) and w not in _ING_ED_NOUNS
   824:                 and not w.endswith("ed")):
   825:             seen_verb = True
   826:     return out

   829: def _runs_from_tags(tags: List[Tuple[str, str]]) -> List[List[Tuple[str, str]]]:
   830:     runs: List[List[Tuple[str, str]]] = []
   831:     cur: List[Tuple[str, str]] = []
   832:     for w, t in tags:
   833:         if _np_eligible(w, t):
   834:             cur.append((w, t))
   835:         else:
   836:             if cur:
   837:                 runs.append(cur)
   838:             cur = []
   839:     if cur:
   840:         runs.append(cur)
   841:     return runs

   844: def _windows_from_run(run: List[Tuple[str, str]]) -> List[Cand]:
   845:     """Every contiguous sub-window that ends on a noun head and survives the
   846:     completeness check. This is what yields both the full compound
   847:     ('weekly strength training') and its sub-anchors ('strength training')."""
   848:     out: List[Cand] = []
   849:     m = len(run)
   850:     for e in range(m):
   851:         we, te = run[e]
   852:         if not _head_is_noun_like(we, te):
   853:             continue
   854:         for s in range(e):                       # window length >= 2
   855:             if e - s + 1 > 6:
   856:                 continue
   857:             toks = [w for w, _ in run[s:e + 1]]
   858:             tgs = [t for _, t in run[s:e + 1]]
   859:             tt, tg = _trim_with_tags(toks, tgs)
   860:             if _is_complete_phrase(tt, tg):
   861:                 out.append((" ".join(tt), "np", tg))
   862:     return out

   876: def _intent_phrases(sent: str) -> List[Cand]:
   877:     s = canonical_phrase(sent)
   878:     out: List[Cand] = []
   879:     for start in INTENT_STARTS:
   880:         idx = s.find(start)
   881:         if idx == -1:
   882:             continue
   883:         # ensure it's a word boundary start
   884:         if idx > 0 and s[idx - 1].isalnum():
   885:             continue
   886:         tail = tokenize(s[idx:])
   887:         cand: List[str] = []
   888:         nstart = len(start.split())
   889:         for t in tail:
   890:             # stop at a clause boundary or a trailing prepositional phrase once
   891:             # the core anchor is captured ("...track cycle length | over time")
   892:             if len(cand) >= nstart and (
   893:                 t in CONJUNCTIONS or t in AUX_VERBS
   894:                 or (t in PREPOSITIONS and t != "of")
   895:             ):
   896:                 break
   897:             cand.append(t)
   898:             if len(cand) >= 8:
   899:                 break
   900:         # drop interior possessives/determiners ("calculate your due date")
   901:         if len(cand) > nstart:
   902:             cand = cand[:nstart] + [t for t in cand[nstart:] if t not in STRIP_INTERIOR]
   903:         while cand and (cand[-1] in TRAILING_BAN or cand[-1] in AUX_VERBS

   927: # ---------------------------------------------------------------------------
   928: # Lightweight, transparent, niche-agnostic quality scoring
   929: # ---------------------------------------------------------------------------

   931: SOURCE_WEIGHT: Dict[str, float] = {
   932:     "title": 0.20, "heading_h1": 0.18, "heading_h2": 0.16, "heading_h3": 0.14,
   933:     "heading_h4": 0.12, "heading_h5": 0.12, "heading_h6": 0.12,
   934:     "list_item": 0.10, "intent": 0.14, "action_object": 0.12, "noun_phrase": 0.10,
   935: }

   940: def _score_phrase(tokens: Sequence[str], tags: Optional[Sequence[str]],
   941:                   source_type: str, snippet: str) -> Dict[str, Any]:
   942:     toks = list(tokens)
   943:     content = [t for t in toks if t not in FUNCTION_WORDS]
   944:     n_content = max(1, len(content))
   945:     tg = list(tags) if tags and len(tags) == len(toks) else [None] * len(toks)
   946: 
   947:     # headedness: structural first (tagger/morphology), domain list is a bonus.
   948:     head_tok, head_tag = toks[-1], tg[-1]
   949:     if head_tok in GENERIC_WEAK_HEADS:
   950:         head = 0.30
   951:     elif head_tok in UNIVERSAL_HEADS:
   952:         head = 1.0
   953:     elif head_tag in {"NOUN", "NUM"} or _head_is_noun_like(head_tok, head_tag):
   954:         head = 0.80                      # any real noun head — niche-agnostic
   955:     else:
   956:         head = 0.50
   957: 
   958:     # entity density: share of "substantial" content tokens (len>=4) plus a
   959:     # small bonus for known-strong vocab. Length-based => works in any niche.
   960:     strong = sum(1 for t in content
   961:                  if len(t) >= 4 or t in UNIVERSAL_HEADS or t in UNIVERSAL_MODIFIERS)
   962:     density = strong / n_content
   963: 
   964:     # specificity: a modifier + head reads as a real concept.
   965:     has_modifier = (
   966:         len(content) >= 2
   967:         and any((len(t) >= 4) or (t in UNIVERSAL_MODIFIERS) for t in toks[:-1])
   968:     )
   969:     specificity = 0.7 if has_modifier else 0.35
   970: 
   971:     # length sweet spot (2-4 content words)
   972:     length = 1.0 if 2 <= len(content) <= 4 else 0.55
   973: 
   974:     # coverage in source snippet
   975:     snip = set(tokenize(snippet))
   976:     coverage = 1.0 if set(toks).issubset(snip) else 0.6
   977: 
   978:     score = (
   979:         0.30 * head +
   980:         0.22 * density +
   981:         0.18 * specificity +
   982:         0.15 * length +
   983:         0.15 * coverage +
   984:         SOURCE_WEIGHT.get(source_type, 0.08)
   985:     )
   986:     score = max(0.0, min(1.0, round(score, 4)))
   987: 
   988:     return {
   989:         "score": score,
   990:         "threshold": ACCEPT_THRESHOLD,
   991:         "decision": "ACCEPT" if score >= ACCEPT_THRESHOLD else "REJECT",
   992:         "signals": {
   993:             "headedness": round(head, 3),
   994:             "entity_density": round(density, 3),
   995:             "specificity": round(specificity, 3),
   996:             "length": round(length, 3),
   997:             "coverage": round(coverage, 3),
   998:         },
   999:     }

  1023: def _add_candidate(
  1024:     out: List[Dict[str, Any]],
  1025:     seen: Set[str],
  1026:     seen_rejects: Set[str],
  1027:     phrase: str,
  1028:     kind: str,
  1029:     tags: Optional[Sequence[str]],
  1030:     source_type: str,
  1031:     section_id: str,
  1032:     snippet: str,
  1033:     doc_id: str = "",
  1034:     workspace_id: str = "default",
  1035:     vertical: str = "general",
  1036: ) -> None:
  1037:     p = canonical_phrase(phrase)
  1038:     tokens = _trim_boundaries(tokenize(p))
  1039:     # drop interior determiners/possessives ("estimate your due date" -> "estimate due date")
  1040:     if len(tokens) > 2:
  1041:         tokens = [tokens[0]] + [t for t in tokens[1:-1] if t not in STRIP_INTERIOR] + [tokens[-1]]
  1042:         tokens = _trim_boundaries(tokens)
  1043:     p = " ".join(tokens)
  1044:     record = {"phrase": p, "source_type": source_type, "section_id": section_id}
  1045: 
  1046:     # tags coming from the chunker were aligned to the *pre-canonical* tokens;
  1047:     # only trust them if they still align, else recompute.
  1048:     tg: Optional[List[str]] = None
  1049:     if tags and len(tags) == len(tokens):
  1050:         tg = list(tags)
  1051: 
  1052:     is_intent = kind == "intent" or (tokens and tokens[0] in INTENT_FIRST_WORDS
  1053:                                       and " ".join(tokens).startswith(INTENT_STARTS))
  1054:     if is_intent:
  1055:         if tg is None:
  1056:             tg = [t for _, t in pos_tag(tokens)]
  1057:         ok = _is_complete_intent(tokens, tg)
  1058:     else:
  1059:         if tg is None:
  1060:             tg = [t for _, t in pos_tag(tokens)]
  1061:         ok = _is_complete_phrase(tokens, tg)
  1062: 
  1063:     if not ok:
  1064:         _reject(record, "incomplete_or_hanging_phrase", seen_rejects,
  1065:                 workspace_id, doc_id, vertical)
  1066:         return
  1067: 
  1068:     scoring = _score_phrase(tokens, tg, source_type, snippet)
  1069:     if scoring["decision"] != "ACCEPT":
  1070:         _reject(record, "below_quality_threshold", seen_rejects,
  1071:                 workspace_id, doc_id, vertical)
  1072:         return
  1073: 
  1074:     key = f"{source_type}:{p}:{section_id}"
  1075:     if key in seen:
  1076:         return
  1077:     seen.add(key)
  1078: 
  1079:     out.append({
  1080:         "phrase": p,
  1081:         "source_type": source_type,
  1082:         "section_id": section_id,
  1083:         "doc_id": doc_id,
  1084:         "snippet": snippet,
  1085:         "extractor_intelligence": scoring,
  1086:     })

  1089: def _classify_source(phrase: str, kind: str) -> str:
  1090:     first = phrase.split()[0] if phrase else ""
  1091:     if kind == "intent" or (first in INTENT_FIRST_WORDS
  1092:                             and phrase.startswith(INTENT_STARTS)):
  1093:         return "intent"
  1094:     return "noun_phrase"
