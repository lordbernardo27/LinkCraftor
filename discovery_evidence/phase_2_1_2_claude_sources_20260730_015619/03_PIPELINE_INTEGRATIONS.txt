def safe_id(
    value: Any,
) -> str:
    text = str(
        value or ""
    ).strip()

    normalized = "".join(
        character
        if (
            character.isalnum()
            or character in {"-", "_", "."}
        )
        else "_"
        for character in text
    )

    normalized = normalized.strip(
        "._"
    )

    return normalized or "default"


def sha256_file(
    path: Path,
) -> str:
    digest = hashlib.sha256()

    with path.open("rb") as handle:
        for block in iter(
            lambda: handle.read(
                1024 * 1024
            ),
            b"",
        ):
            digest.update(block)

    return digest.hexdigest()


def atomic_write_json(
    path: Path,
    value: Mapping[str, Any],
) -> None:
    path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    descriptor, temporary_name = (
        tempfile.mkstemp(
            prefix=f".{path.name}.",
            suffix=".tmp",
            dir=str(path.parent),
        )
    )

    temporary_path = Path(
        temporary_name
    )

    try:
        with os.fdopen(
            descriptor,
            "w",
            encoding="utf-8",
            newline="\n",
        ) as handle:
            json.dump(
                dict(value),
                handle,
                indent=2,
                ensure_ascii=False,
                sort_keys=True,
            )

            handle.write("\n")
            handle.flush()
            os.fsync(
                handle.fileno()
            )

        os.replace(
            temporary_path,
            path,
        )

    finally:
        temporary_path.unlink(
            missing_ok=True,
        )


def load_json(
    path: Path,
) -> dict[str, Any]:
    value = json.loads(
        path.read_text(
            encoding="utf-8-sig",
        )
    )

    if not isinstance(
        value,
        dict,
    ):
        raise RuntimeError(
            f"Expected JSON object: {path}"
        )

    return value


def read_jsonl(
    path: Path,
) -> list[dict[str, Any]]:
    if not path.is_file():
        return []

    records: list[
        dict[str, Any]
    ] = []

    with path.open(
        "r",
        encoding="utf-8",
    ) as handle:
        for line in handle:
            line = line.strip()

            if not line:
                continue

            try:
                value = json.loads(
                    line
                )
            except json.JSONDecodeError:
                continue

            if isinstance(
                value,
                dict,
            ):
                records.append(
                    value
                )

    return records


def _integer_value(
    value: Any,
) -> int | None:
    try:
        converted = int(value)
    except (
        TypeError,
        ValueError,
    ):
        return None

    if converted < 0:
        return None

    return converted


def _first_integer(
    mapping: Mapping[str, Any],
    names: Iterable[str],
) -> int | None:
    for name in names:
        value = _integer_value(
            mapping.get(name)
        )

        if value is not None:
            return value

    return None

def _acquire_lock(
    path: Path,
) -> int | None:
    path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    try:
        return os.open(
            path,
            os.O_CREAT
            | os.O_EXCL
            | os.O_WRONLY,
        )
    except FileExistsError:
        return None


def _release_lock(
    path: Path,

) -> None:
    if descriptor is not None:
        try:
            os.close(
                descriptor
            )
        except OSError:
            pass

    path.unlink(
        missing_ok=True,
    )


def _all_runtime_jobs(

        job_ledger_path,

        workspace_id,

    ledger = read_jsonl(
        job_ledger_path(
            workspace_id
        )
    )

    combined: list[
        dict[str, Any]
    ] = []

    seen_job_ids: set[str] = set()

    for record in [
        *queued,
        *ledger,
    ]:
        if not isinstance(
            record,
            dict,
        ):
            continue

        job_id = str(
            record.get("job_id")
            or ""
        ).strip()

        if job_id and job_id in seen_job_ids:
            continue

        if job_id:
            seen_job_ids.add(
                job_id
            )

        combined.append(
            record
        )

    return combined

    return None

def _derive_expected_upstream_count(
    *,
    processed_udare_jobs: list[
        Mapping[str, Any]
    ],
    manifest: Mapping[str, Any],
    actual_store_count: int,
) -> int:
    candidates: list[int] = []

    for job in processed_udare_jobs:
        payload = _job_payload(
            job
        )

        metadata = _job_metadata(
            job
        )

        for mapping in (
            payload,
            metadata,
        ):
            value = _first_integer(
                mapping,
                (
                    "expected_upstream_count",
                    "population_count",
                    "source_record_count",
                    "raw_html_record_count",
                ),
            )

            if value is not None:
                candidates.append(
                    value
                )

    manifest_value = _first_integer(
        manifest,
        (
            "expected_upstream_count",
            "source_record_count",
            "population_count",
        ),
    )

    if manifest_value is not None:
        candidates.append(
            manifest_value
        )

    if candidates:
        return max(candidates)

    return actual_store_count


def _derive_population_batch_id(
    processed_udare_jobs: list[
        Mapping[str, Any]
    ],
) -> str:
    for job in reversed(
        processed_udare_jobs
    ):
        payload = _job_payload(
            job
        )

        metadata = _job_metadata(
            job
        )

        value = str(
            job.get("batch_id")
            or metadata.get(
                "population_batch_id"
            )
            or payload.get(
                "correlation_id"
            )
            or ""
        ).strip()

        if value:
            return value

    return ""

def maybe_trigger_website_article_integrity_after_udare_queue_drain(

        Mapping[str, Any]

        Mapping[str, Any]

        job
        for job in processed_jobs
        if _job_type(job)
        == UDARE_JOB_TYPE
    ]

    if not processed_udare_jobs:
        return {
            "ok": True,
            "status": "NOT_APPLICABLE",
            "reason": (
                "No UDARE jobs were processed."
            ),
        }

    remaining_udare_jobs = [
        job
        for job in remaining_jobs
        if (
            _job_type(job)
            == UDARE_JOB_TYPE
            and str(
                job.get("status")
                or "queued"
            ).strip().lower()
            in {
                "queued",
                "registered",
                "running",
                "retrying",
            }
        )
    ]

    if remaining_udare_jobs:
        return {
            "ok": True,
            "status": "WAITING_FOR_UDARE",
            "remaining_udare_jobs": len(
                remaining_udare_jobs
            ),
        }

    workspace_id = safe_id(
        workspace_id
    )

    manifest_path = _manifest_path(
        workspace_id
    )

    if not manifest_path.is_file():
        return {
            "ok": False,
            "status": "BLOCKED",
            "reason": (
                "UDARE Store manifest does not exist."
            ),
            "manifest_path": str(
                manifest_path
            ),
        }

    manifest = load_json(
        manifest_path
    )

    record_count = _first_integer(
        manifest,
        (
            "record_count",
        ),
    )

    article_count = _first_integer(
        manifest,
        (
            "article_document_count",
        ),
    )

    metadata_count = _first_integer(
        manifest,
        (
            "metadata_record_count",
        ),
    )

    if (
        record_count is None
        or article_count is None
        or metadata_count is None
    ):
        return {
            "ok": False,
            "status": "BLOCKED",
            "reason": (
                "UDARE manifest counts are incomplete."
            ),
        }

    if not (
        record_count
        == article_count
        == metadata_count
    ):
        return {
            "ok": False,
            "status": "BLOCKED",
            "reason": (
                "UDARE article and metadata counts "
                "do not reconcile."
            ),
            "record_count": record_count,
            "article_document_count": (
                article_count
            ),
            "metadata_record_count": (
                metadata_count
            ),
        }

    expected_upstream_count = (
        _derive_expected_upstream_count(
            processed_udare_jobs=(
                processed_udare_jobs
            ),
            manifest=manifest,
            actual_store_count=record_count,
        )
    )

    if expected_upstream_count < record_count:
        return {
            "ok": False,
            "status": "BLOCKED",
            "reason": (
                "UDARE Store contains more records than "
                "the expected upstream count."
            ),
            "expected_upstream_count": (
                expected_upstream_count
            ),
            "record_count": record_count,
        }

    deferred_upstream_count = (
        expected_upstream_count
        - record_count
    )

    manifest_sha256 = sha256_file(
        manifest_path
    )

    population_batch_id = (
        _derive_population_batch_id(
            processed_udare_jobs
        )
    )

    run_seed = "|".join(
        (
            workspace_id,
            population_batch_id,
            str(expected_upstream_count),
            str(record_count),
            manifest_sha256,
        )
    )

    integrity_run_id = (
        "waini_"
        + hashlib.sha256(
            run_seed.encode("utf-8")
        ).hexdigest()[:24]
    )

    marker_path = (
        _trigger_marker_path(
            workspace_id
        )
    )

    if marker_path.is_file():
        marker = load_json(
            marker_path
        )

        if (
            marker.get(
                "integrity_run_id"
            )
            == integrity_run_id
            and marker.get("status")
            == "TRIGGERED"
        ):
            existing = (
                _find_existing_pipeline_job(
                    workspace_id=workspace_id,
                    integrity_run_id=(
                        integrity_run_id
                    ),
                    job_type=(
                        JOB_TYPE_STRUCTURE
                    ),
                )
            )

            return {
                "ok": True,
                "status": "ALREADY_TRIGGERED",
                "created": False,
                "integrity_run_id": (
                    integrity_run_id
                ),
                "job_id": (
                    existing.get("job_id")
                    if existing
                    else marker.get(
                        "first_job_id"
                    )
                ),
                "marker_path": str(
                    marker_path
                ),
            }

    lock_path = _trigger_lock_path(
        workspace_id
    )

    lock_descriptor = _acquire_lock(
        lock_path
    )

    if lock_descriptor is None:
        return {
            "ok": True,
            "status": "TRIGGER_LOCKED",
            "created": False,
            "integrity_run_id": (
                integrity_run_id
            ),
        }

    try:
        existing = (
            _find_existing_pipeline_job(
                workspace_id=workspace_id,
                integrity_run_id=(
                    integrity_run_id
                ),
                job_type=(
                    JOB_TYPE_STRUCTURE

        )

        if existing is not None:

                "schema_version": (

                ),
                "status": "TRIGGERED",
                "workspace_id": (
                    workspace_id
                ),
                "integrity_run_id": (
                    integrity_run_id
                ),
                "first_job_id": existing.get(
                    "job_id"

                "expected_upstream_count": (
                    expected_upstream_count
                ),
                "expected_store_count": (
                    record_count
                ),
                "deferred_upstream_count": (
                    deferred_upstream_count
                ),
                "udare_manifest_path": str(
                    manifest_path

                "udare_manifest_sha256": (
                    manifest_sha256
                ),

            processed_udare_jobs[-1]

            final_udare_job
        )

        user_id = str(
            final_udare_job.get(
                "user_id"
            )
            or final_payload.get(
                "user_id"
            )
            or "system"
        ).strip() or "system"

            final_udare_job.get(
                "product_id"
            )
            or final_payload.get(
                "product_id"
            )
            or "linkcraftor"
        ).strip() or "linkcraftor"

        failed_execution_count = sum(
            1
            for result in execution_results

                    Mapping,
                )
                and result.get("ok")
                is not True
            )

            "schema_version": (

            ),
            "operation": "execute",
            "auto_continue": True,
            "project_root": str(
                PROJECT_ROOT
            ),
            "workspace_id": workspace_id,
            "integrity_run_id": (
                integrity_run_id
            ),
            "expected_store_count": (
                record_count
            ),
            "expected_store_count_before": (
                record_count
            ),
            "expected_assessed_count": (
                record_count
            ),
            "expected_upstream_count": (
                expected_upstream_count
            ),
            "deferred_upstream_count": (
                deferred_upstream_count
            ),
            "udare_manifest_path": str(
                manifest_path
            ),
            "udare_manifest_sha256": (
                manifest_sha256
            ),
            "payload_ref": str(
                manifest_path
            ),
            "trigger": {
                "type": (
                    "udare_queue_drained"
                ),
                "source_pipeline": (
                    "website_reconstruction"
                ),
                "source_stage": (
                    "udare_reconstruction"
                ),
                "population_batch_id": (
                    population_batch_id
                ),
                "processed_final_batch_count": (
                    len(
                        processed_udare_jobs
                    )
                ),
                "failed_final_batch_count": (
                    failed_execution_count
                ),
                "triggered_at": utc_now(),
            },
            "max_attempts": 3,
        }

        created = (
            _create_pipeline_job_once(
                workspace_id=workspace_id,
                integrity_run_id=(
                    integrity_run_id
                ),
                job_type=(
                    JOB_TYPE_STRUCTURE
                ),
                stage=STAGE_STRUCTURE,
                payload=payload,
                parent_job_id=str(
                    final_udare_job.get(
                        "job_id"
                    )
                    or ""
                ),
                batch_id=(
                    population_batch_id
                    or integrity_run_id
                ),
                user_id=user_id,
                product_id=product_id,
                priority=4,
            )
        )

        marker = {
            "schema_version": (
                AUTOMATION_VERSION
            ),
            "status": "TRIGGERED",
            "workspace_id": workspace_id,
            "integrity_run_id": (
                integrity_run_id
            ),
            "first_job_id": created.get(
                "job_id"
            ),
            "created": created.get(
                "created"
            ),
            "expected_upstream_count": (
                expected_upstream_count
            ),
            "expected_store_count": (
                record_count
            ),
            "deferred_upstream_count": (
                deferred_upstream_count
            ),
            "udare_manifest_path": str(
                manifest_path
            ),
            "udare_manifest_sha256": (
                manifest_sha256
            ),
            "population_batch_id": (
                population_batch_id
            ),
            "triggered_at": utc_now(),
        }

        atomic_write_json(
            marker_path,

def _derive_report_counts(
    document: Mapping[str, Any],
    payload: Mapping[str, Any],

    summary_value = document.get(
        "summary",
        {},
    )

    summary = (

    assessed = _first_integer(

            "articles_assessed",

    )

    if assessed is None:
        assessed = _first_integer(
            payload,
            (

                "expected_assessed_count",

    failed = _first_integer(

            "overall_fail_count",
            "failed_count",
            "articles_failed",

        (
            "overall_pass_count",
            "pass_count",
            "articles_passed",
            "active_pass_count",

    )

    if failed is None:
        failed = _first_integer(
            document,

                "quarantine_candidate_count",
                "distinct_failed_article_count",
            ),
        )

    if (
        passed is None
        and assessed is not None
        and failed is not None
    ):
        passed = assessed - failed

    if (
        failed is None
        and assessed is not None
        and passed is not None

        failed = assessed - passed

    if (
        assessed is None
        or passed is None
        or failed is None
    ):
        raise RuntimeError(
            "Unable to derive quarantine counts "
            "from the Website Integrity Report."
        )

        raise RuntimeError(
            "Website Integrity Report PASS and FAIL "
            "counts do not reconcile."
        )

    return {
        "expected_store_count_before": (
            assessed
        ),
        "expected_assessed_count": (
            assessed
        ),
        "expected_active_count_after": (

            else {}
        )

            == "CERTIFIED"
            and coverage.get(
                "active_certified_count"

            )
            and coverage.get(
                "quarantined_count"
            )
            == _required_integer(
                payload,
                "expected_quarantine_count",
            )
        ):

                "artifact_path": str(path),
                "document": document,
            }

    return None

class ArticleValidationRuntimeRegistrationError(
    RuntimeError
):
    """Raised when Article Validation runtime registration is invalid."""


def _safe_name(
    value: Any,
    *,
    field_name: str,
) -> str:
    normalized = str(
        value or ""
    ).strip()

    if not normalized:
        raise ArticleValidationRuntimeRegistrationError(
            f"{field_name} is required."
        )

    if not re.fullmatch(
        r"[A-Za-z0-9_.:-]+",
        normalized,
    ):
        raise ArticleValidationRuntimeRegistrationError(
            f"{field_name} contains unsupported characters."
        )

    return normalized


def _positive_integer(
    value: Any,
    *,
    field_name: str,
) -> int:
    try:
        normalized = int(
            value
        )

    except (
        TypeError,
        ValueError,
    ) as exc:
        raise ArticleValidationRuntimeRegistrationError(
            f"{field_name} must be an integer."
        ) from exc

    if normalized <= 0:
        raise ArticleValidationRuntimeRegistrationError(
            f"{field_name} must be greater than zero."
        )

    return normalized


def _job_payload(
    job: Mapping[str, Any],

    payload: dict[str, Any] = {}

    nested_payload = job.get(
        "payload"
    )

    if isinstance(
        nested_payload,
        Mapping,

        payload.update(

        )

    for field_name in (
        "workspace_id",
        "expected_active_count",
        "run_id",
        "batch_size",
        "payload_ref",
    ):
        if (
            field_name not in payload
            and field_name in job
        ):
            payload[
                field_name
            ] = job[
                field_name
            ]

    return payload

        Mapping,
    ):
        raise ArticleValidationRuntimeRegistrationError(
            "Article Validation runtime job must be a mapping."
        )

    payload = _job_payload(
        job
    )

    workspace_id = _safe_name(
        payload.get(

        ),
        field_name="workspace_id",

            != certified_active_count

                "Expected active count does not match "

                f"{expected_active_count} != "

        payload.get(

def _normalize_domain(value: str | None) -> str:
    raw = str(value or "").strip().lower()

    if not raw:
        raise HTTPException(
            status_code=400,
            detail="domain or url is required",
        )

    raw = re.sub(r"^https?://", "", raw, flags=re.IGNORECASE)
    raw = raw.split("/", 1)[0].strip()
    raw = raw.rstrip(".")

    if raw.startswith("www."):
        raw = raw[4:]

    if not raw or "." not in raw:
        raise HTTPException(
            status_code=400,
            detail="Invalid domain",
        )

    return raw

    return f"ws_{normalized}"

def _call_by_signature_v1(
    function: Callable[..., Any],
    values: Mapping[str, Any],
) -> Any:
    """
    Invoke a runtime dependency using only parameters accepted by its
    active signature. This protects the worker from minor interface
    differences between runtime adapters.
    """

    signature = inspect.signature(
        function
    )

    positional_arguments: list[Any] = []
    keyword_arguments: Dict[str, Any] = {}

    consumed: set[str] = set()
    accepts_extra_keywords = False

    for parameter in (
        signature.parameters.values()
    ):
        if (
            parameter.kind
            == inspect.Parameter.VAR_POSITIONAL
        ):
            continue

        if (
            parameter.kind
            == inspect.Parameter.VAR_KEYWORD
        ):
            accepts_extra_keywords = True
            continue

        name = parameter.name

        if name in values:
            value = values[
                name
            ]

            consumed.add(
                name
            )

        elif (
            parameter.default
            is not inspect.Parameter.empty
        ):
            continue

        else:
            raise UdareWorkerError(
                "Missing runtime dependency parameter "
                f"{name!r} for {function!r}."
            )

        if (
            parameter.kind
            == inspect.Parameter.POSITIONAL_ONLY
        ):
            positional_arguments.append(
                value
            )

        else:
            keyword_arguments[
                name
            ] = value

    if accepts_extra_keywords:
        for name, value in values.items():
            if name not in consumed:
                keyword_arguments[
                    name
                ] = value

    return function(
        *positional_arguments,
        **keyword_arguments,
    )

) -> str:
    for field in fields:
        value = source.get(
            field
        )

        if value is None:
            continue

        text = str(
            value
        )

        if text.strip():
            return text

    return ""


def _heading_values_v1(
    reconstruction: Mapping[str, Any],
) -> list[str]:
    values: list[str] = []

    raw_headings = reconstruction.get(
        "headings"
    )

    if not isinstance(
        raw_headings,
        list,
    ):
        return values

    for heading in raw_headings:
        if isinstance(
            heading,
            Mapping,
        ):
            text = str(
                heading.get(
                    "text"
                )
                or ""
            ).strip()

        else:
            text = str(
                heading or ""
            ).strip()

        if text:
            values.append(
                text
            )

    return values

                ),

    )

def run_udare_reconstruction_job_v1(
    *,
    job: Mapping[str, Any],
    raw_record_loader: Callable[..., Any] | None = None,
    reconstruction_engine: Callable[..., Any] | None = None,
    document_builder: Callable[..., Any] | None = None,
    store_persister: Callable[..., Any] | None = None,
    manifest_refresher: Callable[..., Any] | None = None,
    status_updater: Callable[..., Any] | None = None,
    progress_updater: Callable[..., Any] | None = None,
    failure_recorder: Callable[..., Any] | None = None,
    concurrency_checker: Callable[..., Any] | None = None,
) -> Dict[str, Any]:
    """
    Execute exactly one authorized UDARE reconstruction job.

    The handler reads one Raw HTML record, reconstructs it once,
    builds one complete HTML reader document, and persists that
    document to the UDARE Store.

    Batch population and queue creation are outside this function.
    """

    job_mapping = _as_mapping_v1(
        job,
        name="job",
    )

    job_type = str(
        job_mapping.get(
            "job_type"
        )
        or job_mapping.get(
            "stage"
        )
        or ""
    ).strip()

    if job_type != UDARE_JOB_TYPE:
        raise UdareWorkerError(
            "The UDARE worker received an unsupported job type: "
            + repr(
                job_type
            )
        )

    payload = _as_mapping_v1(
        job_mapping.get(
            "payload"
        )
        or {},
        name="job payload",
    )

    workspace_id = str(
        job_mapping.get(
            "workspace_id"
        )
        or payload.get(
            "workspace_id"
        )
        or ""
    ).strip()

    job_id = str(
        job_mapping.get(
            "job_id"
        )
        or ""
    ).strip()

    source_record_id = str(
        payload.get(
            "source_record_id"
        )
        or ""
    ).strip()

    html_id = str(
        payload.get(
            "html_id"
        )
        or ""
    ).strip()

    payload_source_url = str(
        payload.get(
            "source_url"
        )
        or ""
    ).strip()

    required_identity = {
        "workspace_id":
            workspace_id,

        "job_id":
            job_id,

        "source_record_id":
            source_record_id,

        "html_id":
            html_id,

        "source_url":
            payload_source_url,
    }

    missing_identity = [
        name

        for name, value
        in required_identity.items()

        if not value
    ]

    if missing_identity:
        raise UdareWorkerError(
            "Incomplete UDARE job identity: "
            + ", ".join(
                missing_identity
            )
        )

    expected_values = {
        "pipeline": (
            str(
                job_mapping.get(
                    "pipeline"
                )
                or payload.get(
                    "pipeline"
                )
                or UDARE_PIPELINE
            )
        ),

        "stage": (
            str(
                job_mapping.get(
                    "stage"
                )
                or job_type
            )
        ),

        "source_store_version": (
            str(
                payload.get(
                    "source_store_version"
                )
                or ""
            )
        ),

        "udare_engine": (
            str(
                payload.get(
                    "udare_engine"
                )
                or ""
            )
        ),

        "target_store": (
            str(
                payload.get(
                    "target_store"
                )
                or ""
            )
        ),

        "article_document_format": (
            str(
                payload.get(
                    "article_document_format"
                )
                or ""
            )
        ),
    }

    exact_contract = {
        "pipeline":
            UDARE_PIPELINE,

        "stage":
            UDARE_JOB_TYPE,

        "source_store_version":
            UDARE_SOURCE_STORE,

        "udare_engine":
            UDARE_ENGINE,

        "target_store":
            UDARE_TARGET_STORE,

        "article_document_format":
            UDARE_ARTICLE_FORMAT,
    }

    mismatches = [
        (
            f"{name}={expected_values[name]!r}; "
            f"expected {expected!r}"
        )

        for name, expected
        in exact_contract.items()

        if expected_values[
            name
        ]
        != expected
    ]

    if mismatches:
        raise UdareWorkerError(
            "UDARE job contract mismatch: "
            + "; ".join(
                mismatches
            )
        )

    execution_controls = payload.get(
        "execution_controls"
    )

    if not isinstance(
        execution_controls,
        Mapping,
    ):
        raise UdareWorkerError(
            "UDARE execution_controls are missing."
        )

    if (
        execution_controls.get(
            "execute_now"
        )
        is not True
    ):
        raise UdareWorkerError(
            "UDARE execution is not authorized for this job."
        )

    if (
        execution_controls.get(
            "store_population_allowed"
        )
        is not True
    ):
        raise UdareWorkerError(
            "UDARE Store population is not authorized for this job."
        )

    raw_record_loader = (
        raw_record_loader
        or _default_raw_record_loader_v1
    )

    reconstruction_engine = (
        reconstruction_engine
        or _default_reconstruction_engine_v1
    )

    document_builder = (
        document_builder
        or _default_document_builder_v1
    )

    store_persister = (
        store_persister
        or _default_store_persister_v1
    )

    manifest_refresher = (
        manifest_refresher
        or _default_manifest_refresher_v1
    )

    status_updater = (
        status_updater
        or _default_status_updater_v1
    )

    progress_updater = (
        progress_updater
        or _default_progress_updater_v1
    )

    failure_recorder = (
        failure_recorder
        or _default_failure_recorder_v1
    )

    concurrency_checker = (
        concurrency_checker
        or _default_concurrency_checker_v1
    )

    max_running = int(
        execution_controls.get(
            "workspace_max_running"
        )
        or 5
    )

    attempt = int(
        job_mapping.get(
            "attempt_count"
        )
        or job_mapping.get(
            "attempts"
        )
        or 0
    )

    batch_id = str(
        job_mapping.get(
            "batch_id"
        )
        or ""
    )

    try:
        concurrency = _call_by_signature_v1(
            concurrency_checker,
            {
                "workspace_id":
                    workspace_id,

                "tenant_id":
                    workspace_id,

                "max_running":
                    max_running,

                "limit":
                    max_running,
            },
        )

        concurrency_mapping = (
            concurrency
            if isinstance(
                concurrency,
                Mapping,
            )
            else {}
        )

        if (
            concurrency_mapping.get(
                "can_start_new_job"
            )
            is False
        ):
            return {
                "ok":
                    False,

                "status":
                    "throttled",

                "worker":
                    WORKER_NAME,

                "workspace_id":
                    workspace_id,

                "job_id":
                    job_id,

                "concurrency":
                    dict(
                        concurrency_mapping
                    ),
            }

        _safe_status_update_v1(
            status_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            status=
                "running",

            message=
                "UDARE reconstruction worker started.",
        )

        _safe_progress_update_v1(
            progress_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            percent=
                10,

            message=
                "Loading Raw HTML record.",

            step=
                "load_raw_html",
        )

        raw_result = _call_by_signature_v1(
            raw_record_loader,
            {
                "workspace_id":
                    workspace_id,

                "tenant_id":
                    workspace_id,

                "html_id":
                    html_id,

                "record_id":
                    source_record_id,

                "source_record_id":
                    source_record_id,
            },
        )

        if raw_result is None:
            raise UdareWorkerError(
                "Raw HTML record was not found: "
                + source_record_id
            )

        raw_record = _normalize_raw_record_v1(
            raw_result
        )

        raw_html = _first_non_empty_v1(
            raw_record,
            (
                "raw_html",
                "html",
                "html_text",
                "source_html",
                "content",
            ),
        )

        if not raw_html:
            raise UdareWorkerError(
                "Raw HTML record contains no HTML document."
            )

        record_url = _first_non_empty_v1(
            raw_record,
            (
                "source_url",
                "url",
                "canonical_url",
                "final_url",
            ),
        )

        source_url = (
            record_url
            or payload_source_url
        )

        if (
            record_url
            and payload_source_url
            and record_url
            != payload_source_url
        ):
            raise UdareWorkerError(
                "Raw HTML record URL does not match "
                "the queued UDARE payload URL."
            )

        source_title = _first_non_empty_v1(
            raw_record,
            (
                "title",
                "page_title",
                "source_title",
            ),
        )

        _safe_progress_update_v1(
            progress_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            percent=
                35,

            message=
                "Running UDARE v1.7 reconstruction.",

            step=
                "reconstruct_article",
        )

        reconstruction_result = (
            _call_by_signature_v1(
                reconstruction_engine,
                {
                    "html_text":
                        raw_html,

                    "raw_html":
                        raw_html,

                    "source_html":
                        raw_html,

                    "source_url":
                        source_url,

                    "url":
                        source_url,

                    "title":
                        source_title,

                    "metadata": {
                        "workspace_id":
                            workspace_id,

                        "job_id":
                            job_id,

                        "source_record_id":
                            source_record_id,

                        "html_id":
                            html_id,

                        "correlation_id":
                            str(
                                payload.get(
                                    "correlation_id"
                                )
                                or ""
                            ),
                    },
                },
            )
        )

        reconstruction = _as_mapping_v1(
            reconstruction_result,
            name="UDARE reconstruction result",
        )

        if reconstruction.get(
            "ok"
        ) is not True:
            raise UdareWorkerError(
                "UDARE reconstruction returned ok=False."
            )

        if (
            str(
                reconstruction.get(
                    "engine"
                )
                or ""
            )
            != UDARE_ENGINE
        ):
            raise UdareWorkerError(
                "UDARE reconstruction returned an "
                "unexpected engine version."
            )

        document_id = (
            source_record_id
            or html_id
        )

        _safe_progress_update_v1(
            progress_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            percent=
                60,

            message=
                "Building complete UDARE article document.",

            step=
                "build_article_document",
        )

        builder_result = _call_by_signature_v1(
            document_builder,
            {
                "reconstruction":
                    reconstruction,

                "source_url":
                    source_url,

                "url":
                    source_url,

                "document_id":
                    document_id,

                "html_id":
                    html_id,
            },
        )

        built_document = _as_mapping_v1(
            builder_result,
            name="UDARE article-document result",
        )

        if built_document.get(
            "ok"
        ) is not True:
            raise UdareWorkerError(
                "UDARE article-document builder returned ok=False."
            )

        if (
            str(
                built_document.get(
                    "article_document_format"
                )
                or ""
            )
            != UDARE_ARTICLE_FORMAT
        ):
            raise UdareWorkerError(
                "UDARE article-document format mismatch."
            )

        article_document = built_document.get(
            "article_document_bytes"
        )

        if article_document is None:
            article_document = built_document.get(
                "article_document"
            )

        if article_document is None:
            raise UdareWorkerError(
                "The UDARE builder returned no article document."
            )

        _safe_progress_update_v1(
            progress_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            percent=
                80,

            message=
                "Persisting article in the UDARE Store.",

            step=
                "persist_udare_article",
        )

        selected_root = reconstruction.get(
            "selected_root"
        )

        selected_tag = (
            str(
                selected_root.get(
                    "tag"
                )
                or ""
            )
            if isinstance(
                selected_root,
                Mapping,
            )
            else ""
        )

        persistence_result = _call_by_signature_v1(
            store_persister,
            {
                "workspace_id":
                    workspace_id,

                "document_id":
                    document_id,

                "html_id":
                    html_id,

                "source_url":
                    source_url,

                "title":
                    str(
                        reconstruction.get(
                            "title"
                        )
                        or built_document.get(
                            "title"
                        )
                        or source_title
                        or ""
                    ),

                "h1":
                    str(
                        reconstruction.get(
                            "h1"
                        )
                        or built_document.get(
                            "h1"
                        )
                        or ""
                    ),

                "headings":
                    _heading_values_v1(
                        reconstruction
                    ),

                "article_document":
                    article_document,

                "udare_engine":
                    UDARE_ENGINE,

                "source_store_version":
                    UDARE_SOURCE_STORE,

                "source_record_id":
                    source_record_id,

                "reconstruction_status":
                    "completed",

                "selected_tag":
                    selected_tag,

                "udare_article_body_sha256":
                    str(
                        built_document.get(
                            "article_body_sha256"
                        )
                        or ""
                    ),

                "content_block_count":
                    int(
                        built_document.get(
                            "content_block_count"
                        )
                        or len(
                            reconstruction.get(
                                "content_blocks"
                            )
                            or []
                        )
                    ),

                "job_id":
                    job_id,

                "batch_id":
                    batch_id,

                "attempt":
                    attempt,

                "operational_metadata": {
                    "worker":
                        WORKER_NAME,

                    "pipeline":
                        UDARE_PIPELINE,

                    "stage":
                        UDARE_JOB_TYPE,

                    "builder":
                        str(
                            built_document.get(
                                "builder"
                            )
                            or ""
                        ),

                    "correlation_id":
                        str(
                            payload.get(
                                "correlation_id"
                            )
                            or ""
                        ),
                },

                "expected_document_sha256":
                    str(
                        built_document.get(
                            "article_document_sha256"
                        )
                        or ""
                    ),

                "allow_replace":
                    False,
            },
        )

        persisted = _as_mapping_v1(
            persistence_result,
            name="UDARE persistence result",
        )

        if persisted.get(
            "ok"
        ) is not True:
            raise UdareWorkerError(
                "UDARE Store persistence returned ok=False."
            )

        defer_global_artifacts = (
            str(
                os.environ.get(
                    "UDARE_DEFER_GLOBAL_ARTIFACTS",
                    "",
                )
            ).strip().casefold()
            in {
                "1",
                "true",
                "yes",
                "on",
            }
        )

        review_result = build_udare_review_document_v1(
            workspace_id=
                workspace_id,

            metadata_path=
                str(
                    persisted.get(
                        "metadata_path"
                    )
                    or ""
                ),
        )

        if (
            not isinstance(
                review_result,
                dict,
            )
            or review_result.get(
                "ok"
            )
            is not True
        ):
            raise UdareWorkerError(
                "UDARE visual review document generation failed."
            )

        if defer_global_artifacts:
            manifest = {
                "ok":
                    True,

                "record_count":
                    0,

                "deferred":
                    True,
            }

            index_result = {
                "ok":
                    True,

                "index_path":
                    "",

                "article_count":
                    0,

                "deferred":
                    True,
            }

        else:
            manifest_result = _call_by_signature_v1(
                manifest_refresher,
                {
                    "workspace_id":
                        workspace_id,

                    "tenant_id":
                        workspace_id,
                },
            )

            manifest = _as_mapping_v1(
                manifest_result,
                name="UDARE manifest refresh result",
            )

            if manifest.get(
                "ok"
            ) is not True:
                raise UdareWorkerError(
                    "UDARE Store manifest refresh failed."
                )

            index_result = build_udare_store_index_v1(
                workspace_id
            )

            if (
                not isinstance(
                    index_result,
                    dict,
                )
                or index_result.get(
                    "ok"
                )
                is not True
            ):
                raise UdareWorkerError(
                    "UDARE Store index generation failed."
                )

        result_summary = {
            "ok":
                True,

            "status":
                "completed",

            "worker":
                WORKER_NAME,

            "pipeline":
                UDARE_PIPELINE,

            "stage":
                UDARE_JOB_TYPE,

            "workspace_id":
                workspace_id,

            "job_id":
                job_id,

            "document_id":
                document_id,

            "html_id":
                html_id,

            "source_record_id":
                source_record_id,

            "source_url":
                source_url,

            "udare_engine":
                UDARE_ENGINE,

            "article_document_format":
                UDARE_ARTICLE_FORMAT,

            "article_document_sha256":
                str(
                    persisted.get(
                        "document_sha256"
                    )
                    or built_document.get(
                        "article_document_sha256"
                    )
                    or ""
                ),

            "article_path":
                str(
                    persisted.get(
                        "article_path"
                    )
                    or ""
                ),

            "metadata_path":
                str(
                    persisted.get(
                        "metadata_path"
                    )
                    or ""
                ),

            "persistence_status":
                str(
                    persisted.get(
                        "status"
                    )
                    or ""
                ),

            "content_block_count":
                int(
                    built_document.get(
                        "content_block_count"
                    )
                    or 0
                ),

            "manifest_record_count":
                int(
                    manifest.get(
                        "record_count"
                    )
                    or 0
                ),

            "review_path":
                str(
                    review_result.get(
                        "review_path"
                    )
                    or ""
                ),

            "review_format":
                str(
                    review_result.get(
                        "format"
                    )
                    or ""
                ),

            "review_image_count":
                int(
                    review_result.get(
                        "image_count"
                    )
                    or 0
                ),

            "index_path":
                str(
                    index_result.get(
                        "index_path"
                    )
                    or ""
                ),

            "index_article_count":
                int(
                    index_result.get(
                        "article_count"
                    )
                    or 0
                ),

            "global_artifacts_deferred":
                defer_global_artifacts,

            "completed_at":
                _utc_now_v1(),
        }

        _safe_progress_update_v1(
            progress_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            percent=
                100,

            message=
                "UDARE reconstruction job completed.",

            step=
                "completed",
        )

        _safe_status_update_v1(
            status_updater,
            workspace_id=
                workspace_id,

            job_id=
                job_id,

            status=
                "completed",

            message=
                "UDARE reconstruction job completed.",

            result=
                result_summary,
        )

        return result_summary

    except Exception as exc:
        error_message = (
            f"{type(exc).__name__}: {exc}"
        )

        try:
            _call_by_signature_v1(
                failure_recorder,
                {
                    "workspace_id":
                        workspace_id,

                    "tenant_id":
                        workspace_id,

                    "job_id":
                        job_id,

                    "id":
                        job_id,

                    "job_type":
                        UDARE_JOB_TYPE,

                    "stage":
                        UDARE_JOB_TYPE,

                    "error":
                        error_message,

                    "error_info":
                        error_message,

                    "payload":
                        dict(
                            payload
                        ),
                },
            )

        except Exception:
            try:
                _safe_status_update_v1(
                    status_updater,
                    workspace_id=
                        workspace_id,

                    job_id=
                        job_id,

                    status=
                        "failed",

                    message=
                        "UDARE reconstruction job failed.",

                    error=
                        error_message,
                )

            except Exception:
                pass

        raise


def explain_udare_reconstruction_worker_v1(
) -> Dict[str, Any]:
    return {
        "worker":
            WORKER_NAME,

        "job_type":
            UDARE_JOB_TYPE,

        "pipeline":
            UDARE_PIPELINE,

        "source_store":
            UDARE_SOURCE_STORE,

        "engine":
            UDARE_ENGINE,

        "article_document_format":
            UDARE_ARTICLE_FORMAT,

        "target_store":
            UDARE_TARGET_STORE,

        "requires_explicit_execution_authorization":
            True,

        "requires_explicit_store_authorization":
            True,

        "reconstructs_each_job_once":
