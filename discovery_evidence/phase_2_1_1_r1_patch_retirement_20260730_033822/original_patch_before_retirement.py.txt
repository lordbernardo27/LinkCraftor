NEW_TIMESTAMP_FUNCTION = '''def _normalize_timestamp(
    value: Any,
    *,
    name: str,
    violations: list,
) -> Optional[str]:
    """Validate and normalize an ISO-8601 timestamp to canonical UTC."""
    if value is None:
        return None

    if not isinstance(value, str):
        violations.append(
            name + " must be an ISO-8601 string or null"
        )
        return None

    text = value.strip()

    if not text:
        return None

    if not _TIMESTAMP_RE.match(text):
        violations.append(
            name + " must be an ISO-8601 timestamp: " + text
        )
        return text

    parse_text = text

    if parse_text.endswith("Z"):
        parse_text = (
            parse_text[:-1]
            + "+00:00"
        )

    try:
        parsed = datetime.fromisoformat(
            parse_text
        )
    except ValueError:
        violations.append(
            name + " is not a valid calendar timestamp: " + text
        )
        return text

    if parsed.tzinfo is None:
        violations.append(
            name + " must include a timezone offset"
        )
        return text

    normalized = (
        parsed
        .astimezone(timezone.utc)
        .isoformat(
            timespec="microseconds"
        )
        .replace(
            "+00:00",
            "Z",
        )
    )

    return normalized
'''

def sha256_file(
    path: Path,
) -> str:
    digest = hashlib.sha256()

    with path.open(
        "rb"
    ) as handle:
        for chunk in iter(
            lambda: handle.read(
                1024 * 1024
            ),
            b"",
        ):
            digest.update(
                chunk
            )

    return digest.hexdigest()


def replace_once(
    source: str,
    old: str,
    new: str,
    label: str,
) -> str:
    count = source.count(
        old
    )

    if count != 1:
        raise RuntimeError(
            f"{label}: expected exactly one match, "
            f"found {count}."
        )

    return source.replace(
        old,
        new,
        1,
    )

def expect_rejected(
    operation,
    label: str,
) -> None:
    try:
        operation()
    except Exception:
        return

    raise AssertionError(
        label + " was unexpectedly accepted"
    )


def verify_behavior(
    module,
) -> None:
    if (
        module.UNIVERSAL_JOB_CONTRACT_ID
        != "urn:linkcraftor:runtime:universal-job-contract"
    ):
        raise AssertionError(
            "Permanent contract ID mismatch."
        )

    candidate = full_candidate()

    job = module.UniversalJob.from_dict(
        candidate
    )

    reconstructed = (
        module.UniversalJob.from_json(
            job.to_canonical_json()
        )
    )

    if job != reconstructed:
        raise AssertionError(
            "Canonical reconstruction changed the job."
        )

    if (
        job.contract_fingerprint()
        != reconstructed.contract_fingerprint()
    ):
        raise AssertionError(
            "Contract fingerprint is not deterministic."
        )

    if (
        job.content_fingerprint()
        != reconstructed.content_fingerprint()
    ):
        raise AssertionError(
            "Content fingerprint is not deterministic."
        )

    if hash(job) != hash(reconstructed):
        raise AssertionError(
            "Deterministic Python hashing failed."
        )

    if (
        module.UniversalJob.required_fields()
        != module.REQUIRED_UNIVERSAL_JOB_FIELDS
    ):
        raise AssertionError(
            "Required-field introspection mismatch."
        )

    schema = module.UniversalJob.schema()

    if (
        schema["contract_id"]
        != module.UNIVERSAL_JOB_CONTRACT_ID
    ):
        raise AssertionError(
            "Schema contract ID mismatch."
        )

    try:
        schema["contract_id"] = "changed"
    except TypeError:
        pass
    else:
        raise AssertionError(
            "Contract schema is mutable."
        )

    missing_candidate = dict(
        candidate
    )

    missing_candidate.pop(
        "job_type"
    )

    expect_rejected(
        lambda: module.UniversalJob.from_dict(
            missing_candidate
        ),
        "Missing required field",
    )

    unknown_candidate = dict(
        candidate
    )

    unknown_candidate[
        "legacy_payload"
    ] = {}

    expect_rejected(
        lambda: module.UniversalJob.from_dict(
            unknown_candidate
        ),
        "Unknown field",
    )

    report = (
        module.validate_universal_job(
            missing_candidate
        )
    )

    if report.is_valid:
        raise AssertionError(
            "Missing-field validation report was marked valid."
        )

    normalized_candidate = dict(
        candidate
    )

    normalized_candidate[
        "created_at"
    ] = "2026-07-29T02:00:00+02:00"

    normalized = (
        module.UniversalJob.from_dict(
            normalized_candidate
        )
    )

    if (
        normalized.created_at
        != "2026-07-29T00:00:00.000000Z"
    ):
        raise AssertionError(
            "Timestamp was not normalized to canonical UTC."
        )

    invalid_time_candidate = dict(
        candidate
    )

    invalid_time_candidate[
        "started_at"
    ] = "2026-07-29T02:00:00Z"

    invalid_time_candidate[
        "completed_at"
    ] = "2026-07-29T01:00:00Z"

    expect_rejected(
        lambda: module.UniversalJob.from_dict(
            invalid_time_candidate
        ),
        "Invalid timestamp ordering",
    )


def rollback() -> None:
    if BACKUP.exists():
        shutil.copy2(
            BACKUP,
            TARGET,
        )
