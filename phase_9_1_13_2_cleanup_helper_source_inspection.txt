def _cleanup_executor_backup_root_v1(
    *,
    project_root: str | Path,
    workspace_id: str,
    execution_request_id: str,
) -> Mapping[str, Any]:

    normalized_workspace_id = (
        _require_string(
            workspace_id,
            field_name="workspace_id",
        )
    )

    normalized_execution_request_id = (
        _require_string(
            execution_request_id,
            field_name="execution_request_id",
        )
    )

    safe_backup_root = (
        _resolve_executor_backup_root_v1(
            project_root=project_root,
            workspace_id=normalized_workspace_id,
            execution_request_id=(
                normalized_execution_request_id
            ),
        )
    )

    safe_workspace_backup_root = (
        safe_backup_root.parent.resolve(
            strict=False
        )
    )

    if (
        safe_backup_root
        == safe_workspace_backup_root
    ):
        raise LifecycleRepairExecutorEngineError(
            "Executor backup cleanup may not target "
            "the workspace backup root itself."
        )

    cleanup_required = (
        safe_backup_root.exists()
    )

    if not cleanup_required:

        return _freeze(
            {
                "backup_cleanup_required":
                    False,

                "backup_cleanup_performed":
                    False,

                "backup_cleanup_verified":
                    True,

                "backup_preserved_for_recovery":
                    False,

                "backup_root":
                    str(
                        safe_backup_root
                    ),

                "cleanup_failure_reason":
                    None,
            }
        )

    if not safe_backup_root.is_dir():

        return _freeze(
            {
                "backup_cleanup_required":
                    True,

                "backup_cleanup_performed":
                    False,

                "backup_cleanup_verified":
                    False,

                "backup_preserved_for_recovery":
                    True,

                "backup_root":
                    str(
                        safe_backup_root
                    ),

                "cleanup_failure_reason":
                    (
                        "Executor backup root is not "
                        "a directory."
                    ),
            }
        )

    cleanup_staging_path = (
        safe_backup_root.with_name(
            safe_backup_root.name
            + ".cleanup_pending"
        )
    )

    cleanup_staging_path = (
        _assert_path_within_root_v1(
            candidate_path=(
                cleanup_staging_path
            ),
            allowed_root=(
                safe_workspace_backup_root
            ),
        )
    )

    if (
        cleanup_staging_path
        == safe_workspace_backup_root
    ):
        raise LifecycleRepairExecutorEngineError(
            "Executor backup cleanup staging path "
            "may not be the workspace backup root."
        )

    if cleanup_staging_path.exists():

        return _freeze(
            {
                "backup_cleanup_required":
                    True,

                "backup_cleanup_performed":
                    False,

                "backup_cleanup_verified":
                    False,

                "backup_preserved_for_recovery":
                    True,

                "backup_root":
                    str(
                        safe_backup_root
                    ),

                "cleanup_failure_reason":
                    (
                        "Backup cleanup staging path "
                        "already exists."
                    ),
            }
        )

    cleanup_performed = False

    cleanup_failure_reason = None

    backup_preserved_for_recovery = False

    try:

        safe_backup_root.replace(
            cleanup_staging_path
        )

        try:

            # ---------------------------------------------
            # Windows-safe bounded bottom-up cleanup.
            #
            # Do not use shutil.rmtree here. On the
            # supported Windows runtime it can fail with
            # WinError 145 even though direct child-first
            # deletion succeeds.
            #
            # The staging directory is already constrained
            # to the verified transaction backup boundary.
            # Symlinks are rejected rather than followed.
            # ---------------------------------------------

            cleanup_entries = sorted(
                cleanup_staging_path.rglob("*"),
                key=lambda path: (
                    len(
                        path.parts
                    ),
                    str(
                        path
                    ).lower(),
                ),
                reverse=True,
            )


            for cleanup_entry in cleanup_entries:

                # -----------------------------------------
                # Fail closed on symbolic links.
                #
                # Executor backups are created from regular
                # repair-target files and should never need
                # symbolic-link traversal.
                # -----------------------------------------

                if cleanup_entry.is_symlink():

                    raise LifecycleRepairExecutorEngineError(
                        "Executor backup cleanup refuses "
                        "symbolic links: "
                        + str(
                            cleanup_entry
                        )
                    )


                # -----------------------------------------
                # Independently confirm the entry remains
                # lexically beneath the staging root.
                # -----------------------------------------

                try:

                    cleanup_entry.relative_to(
                        cleanup_staging_path
                    )

                except ValueError as exc:

                    raise LifecycleRepairExecutorEngineError(
                        "Executor backup cleanup entry "
                        "escaped the staging boundary: "
                        + str(
                            cleanup_entry
                        )
                    ) from exc


                if cleanup_entry.is_file():

                    try:

                        cleanup_entry.unlink()

                    except PermissionError:

                        # ---------------------------------
                        # Windows read-only recovery.
                        # This changes attributes only on
                        # the disposable transaction copy.
                        # ---------------------------------

                        cleanup_entry.chmod(
                            0o600
                        )

                        cleanup_entry.unlink()


                elif cleanup_entry.is_dir():

                    cleanup_entry.rmdir()


                else:

                    raise LifecycleRepairExecutorEngineError(
                        "Executor backup cleanup found an "
                        "unsupported filesystem object: "
                        + str(
                            cleanup_entry
                        )
                    )


            cleanup_staging_path.rmdir()

            cleanup_performed = True

        except Exception as exc:

            cleanup_failure_reason = str(
                exc
            )

            # ---------------------------------------------
            # Cleanup failed after atomic staging rename.
            #
            # Restore the backup directory whenever
            # possible so recovery evidence is not lost.
            # ---------------------------------------------

            try:

                if (
                    cleanup_staging_path.exists()
                    and not safe_backup_root.exists()
                ):

                    cleanup_staging_path.replace(
                        safe_backup_root
                    )

            except Exception as restore_exc:

                cleanup_failure_reason = (
                    cleanup_failure_reason
                    + " | BACKUP_RESTORE_FAILED: "
                    + str(
                        restore_exc
                    )
                )

    except Exception as exc:

        cleanup_failure_reason = str(
            exc
        )

    backup_preserved_for_recovery = (
        safe_backup_root.exists()
    )

    cleanup_verified = (
        not safe_backup_root.exists()
        and not cleanup_staging_path.exists()
    )

    return _freeze(
        {
            "backup_cleanup_required":
                True,

            "backup_cleanup_performed":
                cleanup_performed,

            "backup_cleanup_verified":
                cleanup_verified,

            "backup_preserved_for_recovery":
                backup_preserved_for_recovery,

            "backup_root":
                str(
                    safe_backup_root
                ),

            "cleanup_failure_reason":
                cleanup_failure_reason,
        }
    )
