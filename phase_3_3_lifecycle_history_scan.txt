def _freeze(
    value: Any,
) -> Any:

    if isinstance(
        value,
        Mapping,
    ):

        return MappingProxyType(
            {
                str(key): _freeze(item)
                for key, item
                in value.items()
            }
        )

    if isinstance(
        value,
        list,
    ):

        return tuple(
            _freeze(item)
            for item
            in value
        )

    if isinstance(
        value,
        tuple,
    ):

        return tuple(
            _freeze(item)
            for item
            in value
        )

    return value


def _raw_state_evidence(
    value: Any,
) -> Any:

    if isinstance(
        value,
        UniversalWorkflowStatus,
    ):

        return value.value

    if isinstance(
        value,
        str,
    ):

        return value

    return repr(
        value
    )


def _violation(
    *,
    code: str,
    message: str,
    field: str,
    current_state: Any,
    requested_state: Any,
) -> Mapping[
    str,
    Any,
]:

    return MappingProxyType(
        {
            "code": code,
            "message": message,
            "field": field,
            "current_state": (
                _raw_state_evidence(
                    current_state
                )
            ),
            "requested_state": (
                _raw_state_evidence(
                    requested_state
                )
            ),
        }
    )

@dataclass(
    frozen=True,
    slots=True,
)
class TransitionValidationResult:

    current_state: Optional[
        UniversalWorkflowStatus
    ]

    requested_state: Optional[
        UniversalWorkflowStatus
    ]

    is_valid: bool

    violations: Tuple[
        Mapping[
            str,
            Any,
        ],
        ...,
    ]

    checked_rule_count: int

    validation_version: str = (
        TRANSITION_VALIDATION_VERSION
    )

    def __post_init__(
        self,
    ) -> None:

        object.__setattr__(
            self,
            "violations",
            tuple(
                _freeze(item)
                for item
                in self.violations
            ),
        )

    def to_dict(
        self,
    ) -> Mapping[
        str,
        Any,
    ]:

        return MappingProxyType(
            {
                "current_state": (
                    self.current_state.value
                    if self.current_state
                    is not None
                    else None
                ),
                "requested_state": (
                    self.requested_state.value
                    if self.requested_state
                    is not None
                    else None
                ),
                "is_valid":
                    self.is_valid,
                "violations":
                    self.violations,
                "checked_rule_count":
                    self.checked_rule_count,
                "validation_version":
                    self.validation_version,
            }
        )

def validate_workflow_transition(
    current_state: Any,
    requested_state: Any,
) -> TransitionValidationResult:
    """
    Validate one requested lifecycle transition.

    Rules:

    1. current_state must resolve to UniversalWorkflowStatus.
    2. requested_state must resolve to UniversalWorkflowStatus.
    3. current_state -> requested_state must exist in frozen Phase 3.1.

    This function does not mutate workflow state.
    """

    violations = []

    normalized_current: Optional[
        UniversalWorkflowStatus
    ] = None

    normalized_requested: Optional[
        UniversalWorkflowStatus
    ] = None

    checked_rule_count = 0


    # ------------------------------------------------------------------------
    # Rule 1 - current state
    # ------------------------------------------------------------------------

    checked_rule_count += 1

    try:

        normalized_current = (
            coerce_workflow_state(
                current_state
            )
        )

    except (
        UniversalWorkflowContractError,
        ValueError,
        TypeError,
    ):

        violations.append(
            _violation(
                code=(
                    VIOLATION_CURRENT_STATE_INVALID
                ),
                message=(
                    "current_state does not resolve "
                    "to UniversalWorkflowStatus"
                ),
                field="current_state",
                current_state=current_state,
                requested_state=requested_state,
            )
        )

        return TransitionValidationResult(
            current_state=None,
            requested_state=None,
            is_valid=False,
            violations=tuple(
                violations
            ),
            checked_rule_count=(
                checked_rule_count
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 2 - requested state
    # ------------------------------------------------------------------------

    checked_rule_count += 1

    try:

        normalized_requested = (
            coerce_workflow_state(
                requested_state
            )
        )

    except (
        UniversalWorkflowContractError,
        ValueError,
        TypeError,
    ):

        violations.append(
            _violation(
                code=(
                    VIOLATION_REQUESTED_STATE_INVALID
                ),
                message=(
                    "requested_state does not resolve "
                    "to UniversalWorkflowStatus"
                ),
                field="requested_state",
                current_state=(
                    normalized_current
                ),
                requested_state=(
                    requested_state
                ),
            )
        )

        return TransitionValidationResult(
            current_state=(
                normalized_current
            ),
            requested_state=None,
            is_valid=False,
            violations=tuple(
                violations
            ),
            checked_rule_count=(
                checked_rule_count
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 3 - canonical edge
    # ------------------------------------------------------------------------

    checked_rule_count += 1

    if not has_transition_edge(
        normalized_current,
        normalized_requested,
    ):

        violations.append(
            _violation(
                code=(
                    VIOLATION_TRANSITION_NOT_DECLARED
                ),
                message=(
                    "requested lifecycle transition "
                    "is not declared by the frozen "
                    "Phase 3.1 Workflow State Machine"
                ),
                field="transition",
                current_state=(
                    normalized_current
                ),
                requested_state=(
                    normalized_requested
                ),
            )
        )


    return TransitionValidationResult(
        current_state=(
            normalized_current
        ),
        requested_state=(
            normalized_requested
        ),
        is_valid=(
            not violations
        ),
        violations=tuple(
            violations
        ),
        checked_rule_count=(
            checked_rule_count
        ),
    )


# ============================================================================
# 7. Require-style API
# ============================================================================

def require_valid_workflow_transition(
    current_state: Any,
    requested_state: Any,
) -> TransitionValidationResult:
    """
    Require a valid workflow lifecycle transition.

    Returns the immutable validation result when valid.

    Raises InvalidWorkflowTransitionError when invalid.
    """

    result = (
        validate_workflow_transition(
            current_state,
            requested_state,
        )
    )

    if not result.is_valid:

        raise InvalidWorkflowTransitionError(
            result
        )

    return result

SECTION 9 - PHASE 3.3 CANONICAL DESIGN QUESTIONS
================================================================================

1. Should Phase 3.3 perform a lifecycle transition?

Expected:
NO.

That would combine mutation/execution authority with history recording.


2. Should Phase 3.3 decide whether an edge is legal?

Expected:
NO.

That authority is already frozen in:

    3.2 Transition Validation


3. Should Phase 3.3 accept only validated transition information?

Recommended:
YES.

A history entry should describe a transition known to have passed 3.2.


4. Should Lifecycle History create a second transition graph?

NO.


5. Should Lifecycle History define a new workflow-status enum?

NO.


6. Should Lifecycle History persist records to disk/database?

NO.

Durable persistence belongs to:

    8.0 Workflow State Persistence


7. Then what does 3.3 own?

The canonical immutable HISTORY RECORD MODEL and pure history operations.


8. Should one history entry identify the workflow?

YES.

Proposed identity fields:

    event_id
    workflow_id
    correlation_id


9. Should history identify both states?

YES.

    from_state
    to_state


10. Should a history entry contain time evidence?

YES.

Proposed:

    occurred_at

This is evidence supplied when constructing the record.

3.3 should validate/normalize the timestamp, not generate operational
transition timestamps automatically unless explicitly required.


11. Should 3.3 call datetime.now() internally?

Recommended:
NO.

Why:

- deterministic tests
- caller supplies the actual transition time
- avoids 3.3 becoming transition execution authority


12. Should occurred_at require timezone awareness?

YES.

Canonical timestamps throughout LinkCraftor should remain timezone-aware.


13. Should history entry include transition validation version?

YES.

This provides provenance to the exact validator used.


14. Should history entry include Workflow State Machine version?

YES.

This preserves lifecycle-model provenance.


15. Should history entry include metadata?

YES, but immutable.

Proposed:

    metadata


16. Should history entry include reason?

Recommended:
YES as optional evidence.

Potential:

    reason


17. Should history entry include actor/source?

Recommended:
YES as optional provenance.

Potential fields:

    source
    actor_id


18. Should history know Runtime job IDs?

Not as required canonical fields.

Runtime/job-specific details can live in metadata until later integration phases.


19. Should history record invalid transition attempts?

Recommended:
NO in the canonical lifecycle transition history.

3.3 should record accepted lifecycle transitions.

Invalid attempts belong to validation/audit evidence, not canonical state history.


20. Could invalid-attempt auditing exist later?

YES.

That belongs more naturally to:

    10.5 Audit Trail

rather than the canonical lifecycle-state progression history.


21. Should same-state records be allowed?

NO.

Because 3.2 rejects same-state transitions under the frozen graph.


22. Should terminal outgoing history entries exist?

NO.

Because 3.2 rejects them.


23. Should Lifecycle History expose deterministic sorting?

YES.

Likely by:

    occurred_at
    event_id

But 3.3 should not silently reorder records unless the API explicitly
does so.


24. Should Lifecycle History expose append semantics?

YES, but PURE / in-memory collection transformation.

Example:

    append_lifecycle_history(
        history,
        entry
    )

returns a NEW immutable tuple.

It must not mutate the input collection.


25. Should append require chronological order?

Recommended:
YES.

If the history already has entries:

    new.occurred_at >= last.occurred_at

This protects deterministic lifecycle chronology.


26. Should equal timestamps be allowed?

Recommended:
YES if event_id provides deterministic tie identity.

Multiple lifecycle events may legitimately share timestamp precision.


27. Should duplicate event_id be rejected?

YES.


28. Should duplicate exact transition records be rejected?

At minimum duplicate event_id must be rejected.

Broader idempotency authority is Phase 11.2, so do not overreach.


29. Should history verify workflow_id consistency?

YES.

One history sequence should belong to one workflow.


30. Should correlation_id consistency be mandatory?

Recommended:
YES when both entries provide it.


31. Should event IDs be generated by 3.3?

Recommended:
NO.

Caller supplies canonical event_id.

This preserves deterministic behavior and avoids hidden randomness.

NO.

NO.

NO.

NO.

SECTION 15 - HISTORY SEQUENCE RULES
================================================================================

Proposed validation rules for an immutable history sequence:

RULE 1
------
Every item must be LifecycleHistoryEntry.


RULE 2
------
event_id values must be unique.


RULE 3
------
workflow_id must remain identical across the sequence.


RULE 4
------
correlation_id must remain compatible across the sequence.


RULE 5
------
occurred_at must be non-decreasing.


RULE 6
------
For adjacent entries:

    previous.to_state
        ==
    current.from_state

This creates a continuous lifecycle chain.


RULE 7
------
Every recorded from_state -> to_state must be valid under frozen 3.2.


RULE 8
------
Every entry must carry canonical 3.2 validation provenance.


RULE 9
------
Every entry must carry canonical 3.1 state-machine provenance.
