def require_runtime_registration_admission(
    *,
    registration: Mapping[str, Any],
    runtime_job: Mapping[str, Any],
    worker_id: str,
) -> RuntimeRegistrationAdmissionReport:
    if not isinstance(registration, Mapping):
        raise RuntimeRegistrationAdmissionError(
            "Runtime registration must be a mapping."
        )

    if not isinstance(runtime_job, Mapping):
        raise RuntimeRegistrationAdmissionError(
            "Runtime job must be a mapping."
        )

    job_type = _optional_text(
        registration.get("job_type")
    )

    if job_type is None:
        raise RuntimeRegistrationAdmissionError(
            "Runtime registration has no job_type."
        )

    lifecycle_policy = _mapping(
        registration,
        "lifecycle_policy",
    )
    configuration_requirements = _mapping(
        registration,
        "configuration_requirements",
    )
    version_requirements = _mapping(
        registration,
        "runtime_version_requirements",
    )
    feature_flag_key = _optional_text(
        registration.get("feature_flag_key")
    )

    declared_policy = bool(
        lifecycle_policy
        or configuration_requirements
        or version_requirements
        or feature_flag_key
    )

    binding = _current_binding()

    if binding is None:
        if declared_policy:
            raise RuntimeRegistrationAdmissionError(
                "Registration policy authority is not bound."
            )

        return RuntimeRegistrationAdmissionReport(
            schema_version=(
                REGISTRATION_ADMISSION_SCHEMA_VERSION
            ),
            admitted=True,
            job_type=job_type,
            lifecycle_checked=False,
            configuration_checked=False,
            feature_flag_checked=False,
            version_compatibility_checked=False,
            feature_flag_key=None,
            evaluated_contracts=(),
            evidence=MappingProxyType(
                {
                    "reason": "no_policy_declared",
                }
            ),
        )

    configuration, kernel = binding
    evidence: dict[str, Any] = {}

    if lifecycle_policy:
        evidence["lifecycle"] = _require_lifecycle_policy(
            lifecycle_policy,
            configuration=configuration,
            kernel=kernel,
        )

    if configuration_requirements:
        evidence["configuration"] = _require_configuration(
            configuration_requirements,
            configuration=configuration,
        )

    version_evidence = None

    if version_requirements:
        version_evidence = _require_versions(
            version_requirements,
            job_type=job_type,
            kernel=kernel,
        )
        evidence["version_compatibility"] = version_evidence

    if feature_flag_key:
        evidence["feature_flag"] = _require_feature_flag(
            feature_flag_key,
            runtime_job=runtime_job,
            worker_id=worker_id,
            kernel=kernel,
        )

    evaluated_contracts = (
        ()
        if version_evidence is None
        else tuple(
            version_evidence["contracts"]
        )
    )

    return RuntimeRegistrationAdmissionReport(
        schema_version=(
            REGISTRATION_ADMISSION_SCHEMA_VERSION
        ),
        admitted=True,
        job_type=job_type,
        lifecycle_checked=bool(lifecycle_policy),
        configuration_checked=bool(
            configuration_requirements
        ),
        feature_flag_checked=feature_flag_key is not None,
        version_compatibility_checked=bool(
            version_requirements
        ),
        feature_flag_key=feature_flag_key,
        evaluated_contracts=evaluated_contracts,
        evidence=MappingProxyType(evidence),
    )

def prepare_universal_runtime_job_lease(
    *,
    job_id: str,
    registration: Mapping[str, Any],
    lease_seconds: int = 1800,
) -> Mapping[str, Any]:
    from datetime import timedelta
    from uuid import uuid4

    canonical_job_id = str(job_id or "").strip()

    if not canonical_job_id:
        raise UniversalRuntimeJobLeaseGatewayError(
            "job_id must not be empty."
        )

    if (
        isinstance(lease_seconds, bool)
        or not isinstance(lease_seconds, int)
        or lease_seconds < 1
    ):
        raise UniversalRuntimeJobLeaseGatewayError(
            "lease_seconds must be a positive integer."
        )

    worker = _worker_registration_from_mapping(
        registration
    )

    pool = create_universal_worker_pool_from_registrations(
        pool_id=(
            "universal_runtime_pool_"
            + worker.worker_type
        ),
        worker_type=worker.worker_type,
        registrations=(worker,),
    )

    if len(pool.members) != 1:
        raise UniversalRuntimeJobLeaseGatewayError(
            "Eligible worker pool is invalid."
        )

    assignment = assign_universal_worker(
        job_id=canonical_job_id,
        eligible_workers=(worker,),
    )

    if (
        assignment.status
        is not UniversalWorkerAssignmentStatus.ASSIGNED
        or assignment.worker is None
    ):
        raise UniversalRuntimeJobLeaseGatewayError(
            "No eligible worker was assigned."
        )

    now = datetime.now(timezone.utc)

    started = (
        now.isoformat()
        .replace("+00:00", "Z")
    )

    expires = (
        (
            now
            + timedelta(seconds=lease_seconds)
        )
        .isoformat()
        .replace("+00:00", "Z")
    )

    lease = acquire_universal_worker_lease(
        assignment=assignment,
        lease_id="lease_" + uuid4().hex,
        lease_started_at=started,
        lease_expires_at=expires,
        existing_lease=None,
    )

    if lease.job_id != canonical_job_id:
        raise UniversalRuntimeJobLeaseGatewayError(
            "Lease changed canonical job identity."
        )

    return {
        "job_id":
            canonical_job_id,

        "worker_assignment_status":
            str(
                getattr(
                    assignment.status,
                    "value",
                    assignment.status,
                )
            ),

        "worker_assignment_candidate_count":
            assignment.candidate_count,

        "assigned_worker_id":
            assignment.worker.worker_id,

        "assigned_worker_instance_id":
            assignment.worker.worker_instance_id,

        "worker_pool_id":
            pool.pool_id,

        "lease_owner":
            lease.lease_owner,

        "lease_id":
            lease.lease_id,

        "lease_started_at":
            lease.lease_started_at,

        "lease_expires_at":
            lease.lease_expires_at,

        "lease_active":
            True,

        "lease_released":
            False,

        "lease_gateway_version":
            UNIVERSAL_RUNTIME_JOB_LEASE_GATEWAY_VERSION,
    }

def update_job_status(
    job_id: str,
    new_status: str,
    metadata: Optional[Dict[str, Any]] = None,
    error_message: Optional[str] = None,
) -> OrchestrationJob:
    if new_status not in VALID_JOB_STATUSES:
        raise ValueError(f"Invalid job status: {new_status}")

    jobs = load_jobs()
    if job_id not in jobs:
        raise KeyError(f"Job not found: {job_id}")

    job = jobs[job_id]
    old_status = job.status
    job.status = new_status
    job.updated_at = utc_now()

    normalized_metadata = (
        dict(metadata)
        if isinstance(
            metadata,
            dict,
        )
        else {}
    )

    if normalized_metadata:

        existing_metadata = (
            job.metadata
            if isinstance(
                job.metadata,
                dict,
            )
            else {}
        )

        job.metadata = {
            **existing_metadata,
            **normalized_metadata,
        }

    if error_message:
        job.error_message = error_message

        if new_status == JOB_STATUS_FAILED:
            job.progress_percent = min(
                job.progress_percent,
                99.0,
            )

    jobs[job_id] = job

    save_jobs(
        jobs
    )

    append_job_event(
        job_id,
        old_status,
        new_status,
        normalized_metadata,

def create_job(
    workspace_id: str,
    job_type: str,
    payload: Optional[Dict[str, Any]] = None,
    metadata: Optional[Dict[str, Any]] = None,
    priority: int = 5,
    *,
    job_id: Optional[str] = None,
) -> OrchestrationJob:
    """Persist one orchestration job.

    ``job_id`` is optional for backward compatibility.

    When omitted, this function preserves the historical orchestration
    behaviour and allocates a local ``job_*`` identity.

    When supplied, the caller owns the canonical job identity.  This is the
    ingress required by Universal Job / Runtime Registration integrations:
    the orchestration layer MUST persist that exact identity rather than
    minting a second job ID.
    """
    jobs = load_jobs()

    canonical_job_id = str(
        job_id
        or f"job_{uuid4().hex[:16]}"
    ).strip()

    if not canonical_job_id:
        raise ValueError(
            "job_id must be a non-empty string when supplied"
        )

    existing = jobs.get(
        canonical_job_id
    )

    if existing is not None:
        raise ValueError(
            "Job already exists: "
            + canonical_job_id
        )

    job = OrchestrationJob(
        job_id=canonical_job_id,
        workspace_id=workspace_id,
        job_type=job_type,
        payload=payload or {},
        metadata=metadata or {},
        priority=priority,
    )

    jobs[
        canonical_job_id
    ] = job

    save_jobs(
        jobs
    )

    append_job_event(
        canonical_job_id,
        "none",
        job.status,
        {
            "reason": "job_created",
            "caller_supplied_job_id": (
                job_id is not None
            ),
        },
    )

    return job

            return update_job_status(

            )
        return job

    metadata: Dict[str, Any] = {}
    if worker_id:
        metadata["worker_id"] = worker_id
    return update_job_status(job_id, JOB_STATUS_RUNNING, metadata=metadata)

def update_job_status(
    job_id: str,
    new_status: str,
    metadata: Optional[Dict[str, Any]] = None,
    error_message: Optional[str] = None,
) -> OrchestrationJob:
    if new_status not in VALID_JOB_STATUSES:
        raise ValueError(f"Invalid job status: {new_status}")

    jobs = load_jobs()
    if job_id not in jobs:
        raise KeyError(f"Job not found: {job_id}")

    job = jobs[job_id]
    old_status = job.status
    job.status = new_status
    job.updated_at = utc_now()

    normalized_metadata = (
        dict(metadata)
        if isinstance(
            metadata,
            dict,
        )
        else {}
    )

    if normalized_metadata:

        existing_metadata = (
            job.metadata
            if isinstance(
                job.metadata,
                dict,
            )
            else {}
        )

        job.metadata = {
            **existing_metadata,
            **normalized_metadata,
        }

    if error_message:
        job.error_message = error_message

        if new_status == JOB_STATUS_FAILED:
            job.progress_percent = min(
                job.progress_percent,
                99.0,
            )

    save_jobs(
        jobs
    )

    append_job_event(

        old_status,

def _are_job_dependencies_satisfied(
    job: OrchestrationJob,
    *,
    jobs: Dict[str, OrchestrationJob],
) -> bool:
    """
    Return True only when every declared dependency has succeeded.

    Missing, unresolved, terminally unsuccessful or malformed evidence
    fails closed and leaves the target job queued.
    """
    try:
        dependency_ids = (
            _canonical_dependency_job_ids(
                job
            )
        )

        if not dependency_ids:
            return True

        for dependency_id in dependency_ids:
            dependency = jobs.get(
                dependency_id
            )

            if dependency is None:
                return False

            if (
                dependency.workspace_id
                != job.workspace_id
            ):
                return False

            classification = (
                classify_universal_orchestration_dependency_status(
                    _canonical_dependency_status(
                        dependency
                    )
                )
            )

            if (
                classification.value
                != "satisfied"
            ):
                return False

        return True

    except Exception:
        return False

def _apply_workspace_fairness(
    *,
    queued: List[OrchestrationJob],
    jobs: Dict[str, OrchestrationJob],
) -> List[OrchestrationJob]:
    """
    Apply fairness within each canonical priority class.

    Priority classes are never mixed. For each priority class, exactly
    one current head candidate per workspace is supplied to the existing
    Universal Queue Fairness authority.
    """
    if len(queued) < 2:
        return queued

    service_counts = (
        _workspace_service_counts(
            jobs
        )
    )

    priority_groups: Dict[
        int,
        List[OrchestrationJob],
    ] = {}

    for job in queued:
        priority_groups.setdefault(
            _owner_runtime_compatible_queue_priority(
                job.priority
            ),
            [],
        ).append(
            job
        )

    ordered: List[
        OrchestrationJob
    ] = []

    for priority in sorted(
        priority_groups
    ):
        remaining = list(
            priority_groups[
                priority
            ]
        )

        while remaining:
            workspace_heads: Dict[
                str,
                OrchestrationJob,
            ] = {}

            for candidate in remaining:
                workspace_id = str(
                    candidate.workspace_id
                    or ""
                ).strip()

                if not workspace_id:
                    raise ValueError(
                        "Queued job workspace_id must not be blank."
                    )

                if workspace_id not in workspace_heads:
                    workspace_heads[
                        workspace_id
                    ] = candidate

            fairness_candidates = [
                create_universal_queue_fairness_candidate(
                    workspace_id=workspace_id,
                    job_id=head.job_id,
                    priority=_owner_runtime_compatible_queue_priority(
                        head.priority
                    ),
                    created_at=head.created_at,
                    service_count=service_counts.get(
                        workspace_id,
                        0,
                    ),
                )
                for workspace_id, head
                in workspace_heads.items()
            ]

            decision = (
                select_universal_queue_fairness_candidate(
                    candidates=fairness_candidates,
                )
            )

            selected_job = next(
                candidate
                for candidate in remaining
                if (
                    candidate.job_id
                    == decision.selected_job_id
                )
            )

            ordered.append(
                selected_job
            )

            remaining.remove(
                selected_job
            )

            selected_workspace = (
                selected_job.workspace_id
            )

            service_counts[
                selected_workspace
            ] = (
                service_counts.get(
                    selected_workspace,
                    0,
                )
                + 1
            )

    return ordered

def dequeue_job(
    worker_id: str | None = None,
    *,
    worker_registration: Mapping[str, Any] | None = None,
    lease_seconds: int = 1800,
) -> Optional[OrchestrationJob]:
    """Atomically claim and return the highest-priority queued job.

    The returned job has ALREADY been transitioned to JOB_STATUS_RUNNING
    (with claim metadata) under a cross-process lock, so no other worker
    can receive the same job. Returns None when nothing is queued.

    Workers may still call mark_job_running() afterwards; with the
    corrected service.py that call is idempotent.
    """
    with _dequeue_lock():
        recover_interrupted_runtime_jobs()
        from backend.server.runtime.runtime_feature_flag_gateway import (
            runtime_queue_claims_enabled,
        )
        if not runtime_queue_claims_enabled(worker_id=worker_id):
            return None

        for candidate in _queued_jobs():
            metadata: Dict[str, object] = {
                "claimed_by_dequeue": True,
            }

            if worker_id:
                metadata["worker_id"] = worker_id

            if worker_registration is not None:
                lease_fields = (
                    "lease_owner",
                    "lease_id",
                    "lease_started_at",
                    "lease_expires_at",
                )

                existing_lease = {
                    field:
                        candidate.metadata.get(field)
                    for field in lease_fields
                    if (
                        isinstance(
                            candidate.metadata,
                            Mapping,
                        )
                        and candidate.metadata.get(field)
                    )
                }

                if existing_lease:
                    raise RuntimeError(
                        "Queued job already contains lease "
                        "ownership evidence."
                    )

                lease_metadata = (
                    prepare_universal_runtime_job_lease(
                        job_id=candidate.job_id,
                        registration=worker_registration,
                        lease_seconds=lease_seconds,
                    )
                )

                if (
                    worker_id
                    and lease_metadata.get(
                        "assigned_worker_id"
                    )
                    != worker_id
                ):
                    raise RuntimeError(
                        "Assigned worker does not match "
                        "the claiming worker."
                    )

                metadata.update(
                    dict(lease_metadata)
                )

            try:
                claimed = update_job_status(
                    candidate.job_id,
                    JOB_STATUS_RUNNING,
                    metadata=metadata,
                )
            except Exception:
                # Job vanished or store hiccuped between read and write ù
                # skip it and try the next candidate rather than failing
                # the whole dequeue.
                continue
            return claimed
        return None

def mark_job_running(job_id: str, worker_id: str | None = None) -> OrchestrationJob:
    """Transition a job to RUNNING.

    FIX: made idempotent. dequeue_job() now claims jobs by flipping them to
    RUNNING atomically, so by the time the worker calls this the job is
    usually already running. Re-issuing the transition would be harmless in
    most stores but is wasteful and can clobber claim metadata; if the job is
    already RUNNING we just return it (merging in the worker_id if it wasn't
    recorded yet).
    """
    job = get_job(job_id)
    if job is None:
        raise KeyError(f"unknown job: {job_id}")

    if job.status == JOB_STATUS_RUNNING:
        if worker_id and (job.metadata or {}).get("worker_id") != worker_id:
            # Record the worker identity without re-transitioning status.
            return update_job_status(
                job_id, JOB_STATUS_RUNNING, metadata={"worker_id": worker_id}
            )
        return job

    metadata: Dict[str, Any] = {}
    if worker_id:
        metadata["worker_id"] = worker_id
    return update_job_status(job_id, JOB_STATUS_RUNNING, metadata=metadata)
