def _freeze(value: Any) -> Any:
    if isinstance(value, Mapping):
        return MappingProxyType({str(key): _freeze(item) for key, item in value.items()})
    if isinstance(value, (list, tuple)):
        return tuple(_freeze(item) for item in value)
    if isinstance(value, (set, frozenset)):
        return tuple(sorted((_freeze(item) for item in value), key=repr))
    return value


@dataclass(frozen=True, slots=True)
class RuntimeWorkflowSubmissionResult:
    workflow_id: str
    correlation_id: str
    stage_ids: tuple[str, ...]
    job_ids: tuple[str, ...]
    bridge_result: Any
    mapping_result: Any
    submission_result: Any
    evidence: Mapping[str, Any]
    connection_version: str = "runtime_workflow_submission_connection_v1"

    def __post_init__(self) -> None:
        object.__setattr__(self, "stage_ids", tuple(self.stage_ids))
        object.__setattr__(self, "job_ids", tuple(self.job_ids))
        object.__setattr__(self, "evidence", _freeze(self.evidence))
        if not self.workflow_id or not self.correlation_id:
            raise RuntimeCoordinationGatewayError(
                "Workflow submission identity is incomplete."
            )
        if self.bridge_result.workflow_id != self.workflow_id:
            raise RuntimeCoordinationGatewayError(
                "Bridge workflow identity does not match submission identity."
            )
        if self.mapping_result.workflow_id != self.workflow_id:
            raise RuntimeCoordinationGatewayError(
                "Mapping workflow identity does not match submission identity."
            )
        if self.submission_result.workflow_id != self.workflow_id:
            raise RuntimeCoordinationGatewayError(
                "Runtime submission workflow identity does not match."
            )

    def to_dict(self) -> dict[str, Any]:
        return {
            "connection_version": self.connection_version,
            "workflow_id": self.workflow_id,
            "correlation_id": self.correlation_id,
            "stage_ids": list(self.stage_ids),
            "job_ids": list(self.job_ids),
            "bridge": dict(self.bridge_result.to_dict()),
            "mapping": {
                "workflow_id": self.mapping_result.workflow_id,
                "mapping_count": self.mapping_result.mapping_count,
                "stage_ids": list(self.mapping_result.stage_ids),
            },
            "submission": self.submission_result.to_dict(),
            "evidence": dict(self.evidence),
        }


@dataclass(frozen=True, slots=True)
class RuntimeCoordinationGateway:
    project_root: Path
    workflow_root: Path
    workflow_repository: WorkflowRepository
    gateway_version: str = RUNTIME_COORDINATION_GATEWAY_VERSION

    def bridge_execution_plan_to_runtime(self, **kwargs: Any) -> Any:
        return _bridge_execution_plan_to_runtime(**kwargs)

    def submit_runtime_job_mapping(self, **kwargs: Any) -> Any:
        return _submit_runtime_job_mapping(**kwargs)

    def submit_runtime_execution_wave(self, **kwargs: Any) -> Any:
        return _submit_runtime_execution_wave(**kwargs)

    def submit_workflow_execution(
        self,
        *,
        execution_plan: Any,
        stage_references: Mapping[str, Any],
        context: Any,
        dependency_job_ids_by_stage: Mapping[str, Any] | None = None,
        execution_semantics: str = "parallel_eligible",
        submitter: Any = None,
        correlation_registry: Any = None,
        jobs_reader: Any = None,
    ) -> RuntimeWorkflowSubmissionResult:
        bridge_result = _bridge_execution_plan_to_runtime(
            execution_plan=execution_plan,
            stage_references=stage_references,
            context=context,
        )
        mapping_result = _map_runtime_handoffs_to_job_requests(
            bridge_result=bridge_result,
        )
        submission_arguments: dict[str, Any] = {
            "mapping_result": mapping_result,
            "dependency_job_ids_by_stage": dependency_job_ids_by_stage,
            "execution_semantics": execution_semantics,
        }
        if submitter is not None:
            submission_arguments["submitter"] = submitter
        if correlation_registry is not None:
            submission_arguments["correlation_registry"] = correlation_registry
        if jobs_reader is not None:
            submission_arguments["jobs_reader"] = jobs_reader
        submission_result = _submit_runtime_execution_wave(
            **submission_arguments,
        )
        correlation_id = str(getattr(context, "correlation_id", "") or "").strip()
        evidence = {
            "connection": "runtime_workflow_submission_connection_v1",
            "canonical_bridge": "bridge_execution_plan_to_runtime",
            "canonical_mapper": "map_runtime_handoffs_to_job_requests",
            "canonical_submitter": "submit_runtime_execution_wave",
            "canonical_correlation": "WorkflowJobCorrelationRegistry",
            "workflow_repository": type(self.workflow_repository).__name__,
            "direct_handler_dispatch": False,
            "second_queue_created": False,
            "second_registry_created": False,
            "second_submitter_created": False,
        }
        return RuntimeWorkflowSubmissionResult(
            workflow_id=bridge_result.workflow_id,
            correlation_id=correlation_id,
            stage_ids=tuple(mapping_result.stage_ids),
            job_ids=tuple(submission_result.job_ids),
            bridge_result=bridge_result,
            mapping_result=mapping_result,
            submission_result=submission_result,
            evidence=evidence,
        )

    def intake_runtime_terminal_stage_result(self, *, job_id: str, **kwargs: Any) -> Any:
        if "workflow_repository" in kwargs:
            raise RuntimeCoordinationGatewayError(
                "workflow_repository is owned by RuntimeCoordinationGateway."
            )
        return _intake_runtime_terminal_stage_result(
            job_id=job_id,
            workflow_repository=self.workflow_repository,
            **kwargs,
        )

    def snapshot(self) -> Mapping[str, Any]:
        return _freeze(
            {
                "schema_version": "runtime_coordination_gateway_snapshot_v1",
                "gateway_version": self.gateway_version,
                "component_key": RUNTIME_COORDINATION_GATEWAY_KEY,
                "project_root": str(self.project_root),
                "workflow_root": str(self.workflow_root),
                "workflow_registry": workflow_registry_snapshot(),
                "coordinator_registry": coordinator_registry_snapshot(),
                "authorities": (
                    "coordination_runtime_bridge",
                    "runtime_execution_gateway",
                    "workflow_repository",
                    "workflow_registry",
                    "coordinator_registry",
                ),
            }
        )

    def evidence(self) -> Mapping[str, Any]:

        return _freeze(
            {
                "schema_version": "runtime_coordination_gateway_evidence_v1",
                "gateway_version": self.gateway_version,
                "component_key": RUNTIME_COORDINATION_GATEWAY_KEY,
                "workflow_root": str(self.workflow_root),
                "workflow_registry_count": snapshot["workflow_registry"].get("count", 0),
                "coordinator_registry_count": snapshot["coordinator_registry"].get("count", 0),
                "boot_bound": True,
                "second_runtime_created": False,
                "second_registry_created": False,
                "second_queue_created": False,
            }
        )


def register_runtime_coordination_gateway(
    kernel: UniversalRuntimeKernel,
    project_root: Path | str | None = None,
    **_: Any,
) -> RuntimeCoordinationGateway:
    if kernel.has_component(RUNTIME_COORDINATION_GATEWAY_KEY):
        raise RuntimeCoordinationGatewayBindingError(
            "Runtime coordination gateway is already bound to the kernel."
        )
    resolved_root = Path(project_root or Path.cwd()).resolve()
    workflow_root = (
        resolved_root / "backend" / "server" / "data" / "coordination" / "workflows"
    )
    gateway = RuntimeCoordinationGateway(
        project_root=resolved_root,
        workflow_root=workflow_root,
        workflow_repository=WorkflowRepository(root=workflow_root),
    )
    kernel.bind_component(RUNTIME_COORDINATION_GATEWAY_KEY, gateway)
    return gateway

def intake_runtime_terminal_stage_result(
    *,
    job_id: str,
    workflow_repository: _8f_Any,
    correlation_registry: _8f_Any = None,
    job_reader: _8f_Callable[[str], _8f_Any] = _8f_get_orchestration_job,
    events_reader: _8f_Callable[[str], _8f_Any] = _8f_list_job_events,
    completion_intake: _8f_Callable[..., _8f_Any] = _8f_intake_runtime_completion,
    failure_intake: _8f_Callable[..., _8f_Any] = _8f_intake_runtime_failure,
    correlation_resolver: _8f_Callable[..., _8f_Any] = _8f_resolve_workflow_job_correlation,
    result_processor: _8f_Callable[[_8f_Any], _8f_Any] = _8f_process_stage_result,
) -> RuntimeTerminalIntakeGatewayResult:
    """Intake one completed or failed Runtime job without executing it."""
    normalized_job_id = str(job_id or "").strip()
    if not normalized_job_id:
        raise RuntimeTerminalIntakeGatewayError("job_id is required.")

    load_workflow = getattr(workflow_repository, "load_workflow", None)
    if not callable(load_workflow):
        raise RuntimeTerminalIntakeGatewayError(
            "workflow_repository must expose load_workflow()."
        )
    if not callable(job_reader) or not callable(events_reader):
        raise RuntimeTerminalIntakeGatewayError(
            "Canonical job and event readers must be callable."
        )

    terminal_job = job_reader(normalized_job_id)
    if terminal_job is None:
        raise RuntimeTerminalIntakeGatewayError(
            f"Runtime job was not found: {normalized_job_id}"
        )
    observed_job_id = (
        _8f_text(terminal_job, "job_id")
        or _8f_text(terminal_job, "id")
    )
    if observed_job_id != normalized_job_id:
        raise RuntimeTerminalIntakeGatewayError(
            "Runtime job reader returned a conflicting job identity."
        )

    runtime_status = _8f_text(terminal_job, "status").lower()
    if runtime_status not in {"completed", "failed"}:
        raise RuntimeTerminalIntakeGatewayError(
            "Runtime job is not ready for completion/failure intake: "
            f"{runtime_status or '<missing>'}"
        )

    correlation = correlation_resolver(
        job_id=normalized_job_id,
        registry=correlation_registry,
    )
    workflow_id = _8f_text(correlation, "workflow_id")
    workflow = load_workflow(workflow_id)
    if workflow is None:
        raise RuntimeTerminalIntakeGatewayError(
            f"Persisted workflow was not found: {workflow_id}"
        )

    identity_pairs = (
        ("workflow_id", workflow_id),
        ("workspace_id", _8f_text(correlation, "workspace_id")),
        ("correlation_id", _8f_text(correlation, "correlation_id")),
    )
    for field_name, expected in identity_pairs:
        actual = _8f_text(workflow, field_name)
        if not expected or actual != expected:
            raise RuntimeTerminalIntakeGatewayError(
                "Persisted workflow identity conflicts with Runtime "
                f"correlation: {field_name}"
            )

    fixed_job_reader = lambda requested_id: (
        terminal_job if requested_id == normalized_job_id else None
    )
    if runtime_status == "completed":
        stage_result = completion_intake(
            job_id=normalized_job_id,
            registry=correlation_registry,
            completion_reader=fixed_job_reader,
            events_reader=events_reader,
        )
        intake_authority = "intake_runtime_completion"
    else:
        stage_result = failure_intake(
            job_id=normalized_job_id,
            registry=correlation_registry,
            failure_reader=fixed_job_reader,
            events_reader=events_reader,
        )
        intake_authority = "intake_runtime_failure"

    for field_name, expected in (
        ("job_id", normalized_job_id),
        ("workflow_id", workflow_id),
        ("workspace_id", _8f_text(correlation, "workspace_id")),
        ("correlation_id", _8f_text(correlation, "correlation_id")),
        ("stage_id", _8f_text(correlation, "stage_id")),
    ):
        if _8f_text(stage_result, field_name) != expected:
            raise RuntimeTerminalIntakeGatewayError(
                "Stage Result identity conflicts with Runtime "
                f"correlation: {field_name}"
            )

    result_status = _8f_text(stage_result, "status").lower()
    if result_status != runtime_status:
        raise RuntimeTerminalIntakeGatewayError(
            "Stage Result terminal status conflicts with Runtime job."
        )

    processed_result = result_processor(stage_result)
    disposition = _8f_text(processed_result, "disposition").lower()
    if disposition != runtime_status:
        raise RuntimeTerminalIntakeGatewayError(
            "Processed Stage Result disposition conflicts with Runtime job."
        )

    return RuntimeTerminalIntakeGatewayResult(
        job_id=normalized_job_id,
        workflow_id=workflow_id,
        correlation_id=_8f_text(correlation, "correlation_id"),
        stage_id=_8f_text(correlation, "stage_id"),
        runtime_status=runtime_status,
        disposition=disposition,
        stage_result=_8f_snapshot(stage_result),
        processed_result=_8f_snapshot(processed_result),
        evidence={
            "intake_authority": intake_authority,
            "correlation_authority": (
                "resolve_workflow_job_correlation"
            ),
            "workflow_authority": (
                type(workflow_repository).__name__
            ),
            "processor_authority": "process_stage_result",
            "handler_executed": False,
            "job_created": False,
            "queue_claimed": False,
            "workflow_mutated": False,
        },
    )

    """One atomic repository for Universal Workflow instances."""

    def __init__(
        self,
        *,
        root: Path | str,
    ) -> None:
        repository_root = Path(
            root
        ).resolve()

        repository_root.mkdir(
            parents=True,
            exist_ok=True,
        )

        if not repository_root.is_dir():
            raise WorkflowPersistenceError(
                "Workflow repository root is not a directory."
            )

        self._root = repository_root
        self._lock = threading.RLock()

    @property
    def root(self) -> Path:
        return self._root

    def workflow_path(
        self,
        workflow_id: Any,
    ) -> Path:
        normalized = _required_text(
            workflow_id,
            field_name="workflow_id",
        )

        identity_hash = hashlib.sha256(
            normalized.encode("utf-8")
        ).hexdigest()

        return (
            self._root
            / f"workflow_{identity_hash}.json"
        )

    def exists(
        self,
        workflow_id: Any,
    ) -> bool:
        return self.workflow_path(
            workflow_id
        ).is_file()

    def save_workflow(
        self,
        workflow: UniversalWorkflow,
        *,
        expected_current_fingerprint: Optional[str] = None,
    ) -> WorkflowPersistenceEvidence:
        if not isinstance(
            workflow,
            UniversalWorkflow,
        ):
            raise TypeError(
                "workflow must be a UniversalWorkflow."
            )

        validation = workflow.validate()

        if not validation.is_valid:
            raise WorkflowPersistenceIntegrityError(
                "Workflow contract validation failed: "
                + "; ".join(validation.violations)
            )

        workflow_id = _required_text(
            workflow.workflow_id,
            field_name="workflow_id",
        )

        target = self.workflow_path(
            workflow_id
        )

        with self._lock:
            if expected_current_fingerprint is not None:
                expected = _required_text(
                    expected_current_fingerprint,
                    field_name=(
                        "expected_current_fingerprint"
                    ),
                )

                if not target.is_file():
                    raise WorkflowPersistenceConflictError(
                        "Expected an existing workflow, "
                        "but no persisted workflow exists."
                    )

                current = self.load_workflow_record(
                    workflow_id
                )

                if (
                    current.evidence
                    .workflow_content_fingerprint
                    != expected
                ):
                    raise WorkflowPersistenceConflictError(
                        "Persisted workflow fingerprint "
                        "does not match the expected value."
                    )

            workflow_payload = (
                workflow.to_canonical_dict()
            )

            content_fingerprint = (
                workflow.content_fingerprint()
            )

            identity_fingerprint = (
                workflow.identity_fingerprint()
            )

            persisted_at = _utc_now()

            document_without_fingerprint = {
                "schema_version": (
                    WORKFLOW_PERSISTENCE_DOCUMENT_SCHEMA_VERSION
                ),
                "persistence_version": (
                    WORKFLOW_PERSISTENCE_VERSION
                ),
                "workflow_id": workflow_id,
                "workflow_contract_version": (
                    workflow.contract_version
                ),
                "workflow_content_fingerprint": (
                    content_fingerprint
                ),
                "workflow_identity_fingerprint": (
                    identity_fingerprint
                ),
                "persisted_at": persisted_at,
                "workflow": workflow_payload,
            }

            document_fingerprint = _sha256_payload(
                document_without_fingerprint
            )

            document = dict(
                document_without_fingerprint
            )

            document["document_fingerprint"] = (
                document_fingerprint
            )

            payload = (
                json.dumps(
                    document,
                    ensure_ascii=False,
                    sort_keys=True,
                    indent=2,
                )
                + "\n"
            )

            descriptor, temporary_name = (
                tempfile.mkstemp(
                    dir=str(self._root),
                    prefix=".workflow_",
                    suffix=".tmp",
                )
            )

            temporary = Path(
                temporary_name
            )

            try:
                with os.fdopen(
                    descriptor,
                    "w",
                    encoding="utf-8",
                    newline="\n",
                ) as stream:
                    stream.write(payload)
                    stream.flush()
                    os.fsync(
                        stream.fileno()
                    )

                os.replace(
                    temporary,
                    target,
                )

                try:
                    directory_descriptor = os.open(
                        str(self._root),
                        os.O_RDONLY,
                    )
                except OSError:
                    directory_descriptor = None

                if directory_descriptor is not None:
                    try:
                        os.fsync(
                            directory_descriptor
                        )
                    except OSError:
                        pass
                    finally:
                        os.close(
                            directory_descriptor
                        )

            finally:
                if temporary.exists():
                    temporary.unlink()

            return WorkflowPersistenceEvidence(
                workflow_id=workflow_id,
                path=str(target),
                workflow_content_fingerprint=(
                    content_fingerprint
                ),
                workflow_identity_fingerprint=(
                    identity_fingerprint
                ),
                document_fingerprint=(
                    document_fingerprint
                ),
                persisted_at=persisted_at,
            )

def validate_workflow_transition(
    current_state: Any,
    requested_state: Any,
) -> TransitionValidationResult:
    """
    Validate one requested lifecycle transition.

    Rules:

    1. current_state must resolve to UniversalWorkflowStatus.
    2. requested_state must resolve to UniversalWorkflowStatus.
    3. current_state -> requested_state must exist in frozen Phase 3.1.

    This function does not mutate workflow state.
    """

    violations = []

    normalized_current: Optional[
        UniversalWorkflowStatus
    ] = None

    normalized_requested: Optional[
        UniversalWorkflowStatus
    ] = None

    checked_rule_count = 0


    # ------------------------------------------------------------------------
    # Rule 1 - current state
    # ------------------------------------------------------------------------

    checked_rule_count += 1

    try:

        normalized_current = (
            coerce_workflow_state(
                current_state
            )
        )

    except (
        UniversalWorkflowContractError,
        ValueError,
        TypeError,
    ):

        violations.append(
            _violation(
                code=(
                    VIOLATION_CURRENT_STATE_INVALID
                ),
                message=(
                    "current_state does not resolve "
                    "to UniversalWorkflowStatus"
                ),
                field="current_state",
                current_state=current_state,
                requested_state=requested_state,
            )
        )

        return TransitionValidationResult(
            current_state=None,
            requested_state=None,
            is_valid=False,
            violations=tuple(
                violations
            ),
            checked_rule_count=(
                checked_rule_count
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 2 - requested state
    # ------------------------------------------------------------------------

    checked_rule_count += 1

    try:

        normalized_requested = (
            coerce_workflow_state(
                requested_state
            )
        )

    except (
        UniversalWorkflowContractError,
        ValueError,
        TypeError,
    ):

        violations.append(
            _violation(
                code=(
                    VIOLATION_REQUESTED_STATE_INVALID
                ),
                message=(
                    "requested_state does not resolve "
                    "to UniversalWorkflowStatus"
                ),
                field="requested_state",
                current_state=(
                    normalized_current
                ),
                requested_state=(
                    requested_state
                ),
            )
        )

        return TransitionValidationResult(
            current_state=(
                normalized_current
            ),
            requested_state=None,
            is_valid=False,
            violations=tuple(
                violations
            ),
            checked_rule_count=(
                checked_rule_count
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 3 - canonical edge
    # ------------------------------------------------------------------------

    checked_rule_count += 1

    if not has_transition_edge(
        normalized_current,
        normalized_requested,
    ):

        violations.append(
            _violation(
                code=(
                    VIOLATION_TRANSITION_NOT_DECLARED
                ),
                message=(
                    "requested lifecycle transition "
                    "is not declared by the frozen "
                    "Phase 3.1 Workflow State Machine"
                ),
                field="transition",
                current_state=(
                    normalized_current
                ),
                requested_state=(
                    normalized_requested
                ),
            )
        )


    return TransitionValidationResult(
        current_state=(
            normalized_current
        ),
        requested_state=(
            normalized_requested
        ),
        is_valid=(
            not violations
        ),
        violations=tuple(
            violations
        ),
        checked_rule_count=(
            checked_rule_count
        ),
    )

def validate_terminal_state_mutation(
    current_state: Any,
    requested_state: Any,
) -> TerminalStateProtectionResult:
    """
    Validate a requested lifecycle mutation through the terminal guard.

    Processing order:

    1. Normalize current state.
    2. Normalize requested state.
    3. If current state is terminal, prohibit mutation immediately.
    4. Otherwise delegate edge legality to frozen Phase 3.2.

    No workflow mutation occurs here.
    """

    # ------------------------------------------------------------------------
    # Rule 1 - current-state normalization
    # ------------------------------------------------------------------------

    try:

        normalized_current = (
            coerce_workflow_state(
                current_state
            )
        )

    except (
        UniversalWorkflowContractError,
        ValueError,
        TypeError,
    ):

        return _result(
            current_state=None,
            requested_state=None,
            current_state_is_terminal=False,
            mutation_allowed=False,
            protection_triggered=False,
            code=(
                VIOLATION_CURRENT_STATE_INVALID
            ),
            reason=(
                "current_state does not resolve "
                "to UniversalWorkflowStatus"
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 2 - explicit terminal-state protection
    # ------------------------------------------------------------------------

    if is_terminal_workflow_state(
        normalized_current
    ):

        normalized_requested = None

        try:

            normalized_requested = (
                coerce_workflow_state(
                    requested_state
                )
            )

        except (
            UniversalWorkflowContractError,
            ValueError,
            TypeError,
        ):

            pass

        return _result(
            current_state=(
                normalized_current
            ),
            requested_state=(
                normalized_requested
            ),
            current_state_is_terminal=True,
            mutation_allowed=False,
            protection_triggered=True,
            code=(
                TERMINAL_STATE_MUTATION_PROHIBITED
            ),
            reason=(
                "workflow has reached a canonical "
                "terminal state; lifecycle mutation "
                "is prohibited"
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 3 - requested-state normalization
    # ------------------------------------------------------------------------

    try:

        normalized_requested = (
            coerce_workflow_state(
                requested_state
            )
        )

    except (
        UniversalWorkflowContractError,
        ValueError,
        TypeError,
    ):

        return _result(
            current_state=(
                normalized_current
            ),
            requested_state=None,
            current_state_is_terminal=False,
            mutation_allowed=False,
            protection_triggered=False,
            code=(
                VIOLATION_REQUESTED_STATE_INVALID
            ),
            reason=(
                "requested_state does not resolve "
                "to UniversalWorkflowStatus"
            ),
        )


    # ------------------------------------------------------------------------
    # Rule 4 - delegate edge legality to frozen Phase 3.2
    # ------------------------------------------------------------------------

    validation = (
        validate_workflow_transition(
            normalized_current,
            normalized_requested,
        )
    )

    if not validation.is_valid:

        violation = (
            validation.violations[
                0
            ]
            if validation.violations
            else None
        )

        code = (
            violation[
                "code"
            ]
            if violation
            is not None
            else VIOLATION_TRANSITION_NOT_DECLARED
        )

        reason = (
            violation[
                "message"
            ]
            if violation
            is not None
            else (
                "requested lifecycle transition "
                "is invalid under frozen Phase 3.2"
            )
        )

        return _result(
            current_state=(
                normalized_current
            ),
            requested_state=(
                normalized_requested
            ),
            current_state_is_terminal=False,
            mutation_allowed=False,
            protection_triggered=False,
            code=code,
            reason=reason,
        )

    return _result(
        current_state=(
            normalized_current
        ),
        requested_state=(
            normalized_requested
        ),
        current_state_is_terminal=False,
        mutation_allowed=True,
        protection_triggered=False,
        code=None,
        reason=(
            "current state is non-terminal and "
            "requested transition is valid under "
            "frozen Phase 3.2"
        ),
    )

def append_lifecycle_history(
    history: Iterable[
        LifecycleHistoryEntry
    ],
    entry: LifecycleHistoryEntry,
) -> Tuple[
    LifecycleHistoryEntry,
    ...,
]:
    """
    Append one entry by returning a new immutable tuple.

    The input history is never mutated.
    """

    if not isinstance(
        entry,
        LifecycleHistoryEntry,
    ):

        raise InvalidLifecycleHistoryEntryError(
            (
                "entry must be "
                "LifecycleHistoryEntry"
            )
        )

    entries = tuple(
        history
    )

    existing = (
        validate_lifecycle_history(
            entries
        )
    )

    if not existing.is_valid:

        raise LifecycleHistorySequenceError(
            (
                "existing lifecycle history "
                f"is invalid: "
                f"{existing.violations!r}"
            )
        )

    if any(
        existing_entry.event_id
        == entry.event_id
        for existing_entry
        in entries
    ):

        raise DuplicateLifecycleHistoryEventError(
            (
                "duplicate lifecycle-history "
                f"event_id: {entry.event_id!r}"
            )
        )

    if entries:

        last = entries[-1]

        if (
            last.workflow_id
            != entry.workflow_id
        ):

            raise LifecycleHistorySequenceError(
                (
                    "workflow_id mismatch between "
                    "history and appended entry"
                )
            )

        if (
            last.correlation_id
            != entry.correlation_id
        ):

            raise LifecycleHistorySequenceError(
                (
                    "correlation_id mismatch between "
                    "history and appended entry"
                )
            )

        if (
            _occurred_at_datetime(
                entry.occurred_at
            )
            <
            _occurred_at_datetime(
                last.occurred_at
            )
        ):

            raise LifecycleHistorySequenceError(
                (
                    "appended occurred_at cannot "
                    "precede the last history event"
                )
            )

        if (
            last.to_state
            != entry.from_state
        ):

            raise LifecycleHistorySequenceError(
                (
                    "appended lifecycle transition "
                    "does not continue from the "
                    "last recorded state"
                )
            )

    candidate = (
        entries
        + (
            entry,
        )
    )

    result = (
        validate_lifecycle_history(
            candidate
        )
    )

    if not result.is_valid:

        raise LifecycleHistorySequenceError(
            (
                "appended lifecycle history "
                f"is invalid: "
                f"{result.violations!r}"
            )
        )

    return candidate
