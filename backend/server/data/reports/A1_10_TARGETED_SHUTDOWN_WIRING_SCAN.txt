  182: class RuntimeBootProcess:
  183:     """
  184:     Compose and start the Universal Runtime foundation.
  185: 
  186:     Composition order:
  187: 
  188:     1. Load Runtime Configuration.
  189:     2. Validate Runtime Environment.
  190:     3. Create Universal Runtime Kernel.
  191:     4. Create and bind Runtime Service Registry.
  192:     5. Execute controlled service registrars.
  193:     6. Create and bind Runtime Lifecycle Manager.
  194:     7. Validate required kernel components.
  195:     8. Seal the service registry.
  196:     9. Initialize services.
  197:     10. Start services.
  198:     11. Publish immutable boot context.
  199: 
  200:     Concurrent boot requests are serialized through one asyncio lock.
  201:     Once running, repeated boot requests return the existing context.
  202:     """
  203:

  208:         "_registrars",

  210:         "_status",

  221:     def __init__(
  222:         self,
  223:         *,
  224:         project_root: Path,
  225:         environ: Mapping[str, str] | None = None,
  226:         overrides: Mapping[str, Any] | None = None,
  227:         service_registrars: Sequence[
  228:             RuntimeServiceRegistrar
  229:         ] = (),
  230:         event_limit: int = 500,
  231:     ) -> None:
  232:         resolved_root = (
  233:             Path(project_root)
  234:             .expanduser()
  235:             .resolve()
  236:         )
  237: 
  238:         if (
  239:             not resolved_root.exists()
  240:             or not resolved_root.is_dir()
  241:         ):
  242:             raise RuntimeBootConfigurationError(
  243:                 "project_root must be an existing "
  244:                 f"directory: {resolved_root}"
  245:             )
  246: 
  247:         if isinstance(event_limit, bool):
  248:             raise RuntimeBootConfigurationError(
  249:                 "event_limit must be an integer."
  250:             )
  251: 
  252:         resolved_event_limit = int(event_limit)
  253: 
  254:         if not 10 <= resolved_event_limit <= 100000:
  255:             raise RuntimeBootConfigurationError(
  256:                 "event_limit must be between 10 and 100000."
  257:             )
  258: 
  259:         registrars = tuple(service_registrars)
  260: 
  261:         invalid_registrar_indexes = tuple(
  262:             index
  263:             for index, registrar in enumerate(registrars)
  264:             if not callable(registrar)
  265:         )
  266: 
  267:         if invalid_registrar_indexes:
  268:             raise RuntimeBootConfigurationError(
  269:                 "All service registrars must be callable. "
  270:                 "Invalid indexes: "
  271:                 + ", ".join(
  272:                     str(index)
  273:                     for index in invalid_registrar_indexes
  274:                 )
  275:             )
  276: 
  277:         self._project_root = resolved_root
  278: 
  279:         self._environ = (
  280:             None
  281:             if environ is None
  282:             else MappingProxyType(
  283:                 dict(environ)
  284:             )
  285:         )
  286: 
  287:         self._overrides = MappingProxyType(
  288:             dict(overrides or {})
  289:         )
  290: 
  291:         self._registrars = registrars
  292:         self._event_limit = resolved_event_limit
  293:         self._status = RuntimeBootStatus.IDLE
  294:         self._attempt = 0
  295:         self._boot_id: str | None = None
  296:         self._stage = "idle"
  297:         self._generation = 0
  298:         self._events: list[RuntimeBootEvent] = []
  299:         self._context: RuntimeBootContext | None = None
  300:         self._failure_reason: str | None = None
  301:         self._lock = asyncio.Lock()
  302:

  313:     def _record_event(
  314:         self,
  315:         *,
  316:         stage: str,
  317:         outcome: str,
  318:         detail: str | None = None,
  319:     ) -> RuntimeBootEvent:
  320:         self._generation += 1
  321:         self._stage = str(stage)

  323:         event = RuntimeBootEvent(
  324:             sequence=self._generation,
  325:             timestamp=_utc_now(),

  327:             stage=self._stage,
  328:             outcome=str(outcome),

  362:     async def _cleanup_failed_boot(
  363:         self,
  364:         *,
  365:         kernel: UniversalRuntimeKernel | None,
  366:         lifecycle_manager: (
  367:             RuntimeLifecycleManager | None
  368:         ),
  369:     ) -> tuple[str, ...]:
  370:         """
  371:         Perform best-effort cleanup after failed composition or startup.
  372:         """
  373: 
  374:         failures: list[str] = []
  375: 
  376:         if (
  377:             lifecycle_manager is not None
  378:             and lifecycle_manager.phase
  379:             not in {
  380:                 RuntimeLifecyclePhase.CREATED,
  381:                 RuntimeLifecyclePhase.STOPPED,
  382:             }
  383:         ):
  384:             try:
  385:                 await lifecycle_manager.stop()
  386:             except Exception as exc:
  387:                 failures.append(
  388:                     "lifecycle_cleanup="
  389:                     f"{type(exc).__name__}: {exc}"
  390:                 )
  391: 
  392:         elif (
  393:             kernel is not None
  394:             and kernel.state.value
  395:             not in {
  396:                 "failed",
  397:                 "stopped",
  398:                 "stopping",
  399:             }
  400:         ):
  401:             try:
  402:                 kernel.mark_failed(
  403:                     self._failure_reason
  404:                     or "runtime boot failure"
  405:                 )
  406:             except Exception as exc:
  407:                 failures.append(
  408:                     "kernel_failure_mark="
  409:                     f"{type(exc).__name__}: {exc}"
  410:                 )
  411: 
  412:         return tuple(failures)

  414:     async def boot(
  415:         self,
  416:     ) -> RuntimeBootContext:
  417:         """
  418:         Compose and start the runtime.
  419: 
  420:         The asyncio lock creates a single-flight boot boundary. A second
  421:         caller waits for the first caller and then receives the same
  422:         successful context.
  423:         """
  424: 
  425:         async with self._lock:
  426:             if (
  427:                 self._status
  428:                 is RuntimeBootStatus.SHUTTING_DOWN
  429:             ):
  430:                 raise RuntimeBootConfigurationError(
  431:                     "Runtime boot is unavailable while "
  432:                     "shutdown is in progress."
  433:                 )
  434: 
  435:             if (
  436:                 self._status
  437:                 is RuntimeBootStatus.RUNNING
  438:                 and self._context is not None
  439:             ):
  440:                 self._record_event(
  441:                     stage="boot",
  442:                     outcome="already_running",
  443:                     detail=self._context.boot_id,
  444:                 )
  445: 
  446:                 return self._context
  447: 
  448:             self._attempt += 1
  449:             self._boot_id = (
  450:                 f"runtime_boot_{uuid4().hex}"
  451:             )
  452:             self._status = RuntimeBootStatus.BOOTING
  453:             self._failure_reason = None
  454:             self._context = None
  455: 
  456:             self._record_event(
  457:                 stage="boot",
  458:                 outcome="started",
  459:                 detail=f"attempt={self._attempt}",
  460:             )
  461: 
  462:             kernel: UniversalRuntimeKernel | None = None
  463: 
  464:             lifecycle_manager: (
  465:                 RuntimeLifecycleManager | None
  466:             ) = None
  467: 
  468:             stage = "configuration"
  469: 
  470:             try:
  471:                 loader = RuntimeConfigurationLoader(
  472:                     project_root=self._project_root
  473:                 )
  474: 
  475:                 configuration = loader.load(
  476:                     environ=self._environ,
  477:                     overrides=self._overrides,
  478:                 )
  479: 
  480:                 self._record_event(
  481:                     stage=stage,
  482:                     outcome="completed",
  483:                     detail=configuration.fingerprint(),
  484:                 )
  485: 
  486:                 stage = "environment"
  487: 
  488:                 environment = RuntimeEnvironmentManager(
  489:                     configuration
  490:                 )
  491: 
  492:                 self._record_event(
  493:                     stage=stage,
  494:                     outcome="completed",
  495:                     detail=environment.name.value,
  496:                 )
  497: 
  498:                 stage = "kernel"
  499: 
  500:                 kernel = UniversalRuntimeKernel(
  501:                     runtime_id=configuration.runtime_id,
  502:                     product_name=configuration.product_name,
  503:                 )
  504: 
  505:                 kernel.bind_component(
  506:                     "configuration",
  507:                     configuration,
  508:                 )
  509: 
  510:                 kernel.bind_component(
  511:                     "environment",
  512:                     environment,
  513:                 )
  514: 
  515:                 self._record_event(
  516:                     stage=stage,
  517:                     outcome="completed",
  518:                     detail=configuration.runtime_id,
  519:                 )
  520: 
  521:                 stage = "service_registry"
  522: 
  523:                 service_registry = (
  524:                     RuntimeServiceRegistry()
  525:                 )
  526: 
  527:                 kernel.bind_component(
  528:                     "service_registry",
  529:                     service_registry,
  530:                 )
  531: 
  532:                 self._record_event(
  533:                     stage=stage,
  534:                     outcome="created",
  535:                 )
  536: 
  537:                 stage = "service_registration"
  538: 
  539:                 composition = RuntimeBootComposition(
  540:                     project_root=self._project_root,
  541:                     configuration=configuration,
  542:                     environment=environment,
  543:                     kernel=kernel,
  544:                     service_registry=service_registry,
  545:                 )
  546: 
  547:                 for registrar in self._registrars:
  548:                     registrar_name = (
  549:                         self._registrar_name(
  550:                             registrar
  551:                         )
  552:                     )
  553: 
  554:                     registrar(composition)
  555: 
  556:                     self._record_event(
  557:                         stage=stage,
  558:                         outcome="registrar_completed",
  559:                         detail=registrar_name,
  560:                     )
  561: 
  562:                 stage = "lifecycle_manager"
  563: 
  564:                 lifecycle_manager = (
  565:                     RuntimeLifecycleManager(
  566:                         kernel=kernel,
  567:                         service_registry=(
  568:                             service_registry
  569:                         ),
  570:                         startup_timeout_seconds=(
  571:                             configuration
  572:                             .startup_timeout_seconds
  573:                         ),
  574:                         shutdown_timeout_seconds=(
  575:                             configuration
  576:                             .shutdown_timeout_seconds
  577:                         ),
  578:                     )
  579:                 )
  580: 
  581:                 kernel.bind_component(
  582:                     "lifecycle_manager",
  583:                     lifecycle_manager,
  584:                 )
  585: 
  586:                 kernel.require_components(
  587:                     list(
  588:                         BOOT_REQUIRED_COMPONENT_KEYS
  589:                     )
  590:                 )
  591: 
  592:                 self._record_event(
  593:                     stage=stage,
  594:                     outcome="completed",
  595:                 )
  596: 
  597:                 stage = "registry_seal"
  598: 
  599:                 registry_snapshot = (
  600:                     service_registry.seal()
  601:                 )
  602: 
  603:                 self._record_event(
  604:                     stage=stage,
  605:                     outcome="completed",
  606:                     detail=(
  607:                         "services="
  608:                         f"{registry_snapshot.service_count}"
  609:                     ),
  610:                 )
  611: 
  612:                 stage = "lifecycle_initialize"
  613: 
  614:                 await lifecycle_manager.initialize()
  615: 
  616:                 self._record_event(
  617:                     stage=stage,
  618:                     outcome="completed",
  619:                 )
  620: 
  621:                 stage = "lifecycle_start"
  622: 
  623:                 await lifecycle_manager.start()
  624: 
  625:                 self._record_event(
  626:                     stage=stage,
  627:                     outcome="completed",
  628:                 )
  629: 
  630:                 stage = "finalize"
  631: 
  632:                 service_keys = (
  633:                     service_registry
  634:                     .dependency_order()
  635:                 )
  636: 
  637:                 context = RuntimeBootContext(
  638:                     boot_id=self._boot_id,
  639:                     attempt=self._attempt,
  640:                     project_root=self._project_root,
  641:                     configuration=configuration,
  642:                     environment=environment,
  643:                     kernel=kernel,
  644:                     service_registry=(
  645:                         service_registry
  646:                     ),
  647:                     lifecycle_manager=(
  648:                         lifecycle_manager
  649:                     ),
  650:                     service_keys=service_keys,
  651:                     booted_at=_utc_now(),
  652:                 )
  653: 
  654:                 self._context = context
  655:                 self._status = (
  656:                     RuntimeBootStatus.RUNNING
  657:                 )
  658: 
  659:                 self._record_event(
  660:                     stage=stage,
  661:                     outcome="completed",
  662:                     detail=self._boot_id,
  663:                 )
  664: 
  665:                 return context
  666: 
  667:             except Exception as exc:
  668:                 self._status = (
  669:                     RuntimeBootStatus.FAILED
  670:                 )
  671: 
  672:                 self._failure_reason = (
  673:                     f"{type(exc).__name__}: {exc}"
  674:                 )
  675: 
  676:                 self._record_event(
  677:                     stage=stage,
  678:                     outcome="failed",
  679:                     detail=self._failure_reason,
  680:                 )
  681: 
  682:                 cleanup_failures = (
  683:                     await self._cleanup_failed_boot(
  684:                         kernel=kernel,
  685:                         lifecycle_manager=(
  686:                             lifecycle_manager
  687:                         ),
  688:                     )
  689:                 )
  690: 
  691:                 if cleanup_failures:
  692:                     self._record_event(
  693:                         stage="failure_cleanup",
  694:                         outcome=(
  695:                             "completed_with_failures"
  696:                         ),
  697:                         detail="; ".join(
  698:                             cleanup_failures
  699:                         ),
  700:                     )
  701:                 else:
  702:                     self._record_event(
  703:                         stage="failure_cleanup",
  704:                         outcome="completed",
  705:                     )
  706: 
  707:                 raise RuntimeBootFailureError(
  708:                     stage=stage,
  709:                     cause=exc,
  710:                 ) from exc
  711: 
  712:     async def begin_shutdown(
  713:         self,
  714:         *,
  715:         context: RuntimeBootContext,
  716:     ) -> None:
  717:         """Lock the boot controller into SHUTTING_DOWN."""
  718: 
  719:         async with self._lock:
  720:             if self._context is not context:
  721:                 raise RuntimeBootConfigurationError(
  722:                     "Shutdown context does not match the "
  723:                     "active runtime boot context."
  724:                 )
  725: 
  726:             if (
  727:                 self._status
  728:                 is RuntimeBootStatus.STOPPED
  729:             ):
  730:                 return
  731: 
  732:             if (
  733:                 self._status
  734:                 is not RuntimeBootStatus.RUNNING
  735:             ):
  736:                 raise RuntimeBootConfigurationError(
  737:                     "Runtime shutdown can begin only while "
  738:                     f"boot status is 'running', not "
  739:                     f"{self._status.value!r}."
  740:                 )
  741: 
  742:             self._status = (
  743:                 RuntimeBootStatus.SHUTTING_DOWN
  744:             )
  745: 
  746:             self._record_event(
  747:                 stage="shutdown",
  748:                 outcome="started",
  749:                 detail=context.boot_id,
  750:             )

  752:     async def complete_shutdown(
  753:         self,
  754:         *,
  755:         context: RuntimeBootContext,
  756:         failure_reason: str | None = None,
  757:     ) -> None:
  758:         """Publish a completed runtime shutdown."""
  759: 
  760:         async with self._lock:
  761:             if self._context is not context:
  762:                 raise RuntimeBootConfigurationError(
  763:                     "Shutdown context does not match the "
  764:                     "active runtime boot context."
  765:                 )
  766: 
  767:             if (
  768:                 context.lifecycle_manager.phase.value
  769:                 != "stopped"
  770:             ):
  771:                 raise RuntimeBootConfigurationError(
  772:                     "Runtime shutdown cannot be completed "
  773:                     "before the lifecycle reaches STOPPED."
  774:                 )
  775: 
  776:             if self._status not in {
  777:                 RuntimeBootStatus.SHUTTING_DOWN,
  778:                 RuntimeBootStatus.STOPPED,
  779:             }:
  780:                 raise RuntimeBootConfigurationError(
  781:                     "Runtime shutdown completion is invalid "
  782:                     f"while boot status is "
  783:                     f"{self._status.value!r}."
  784:                 )
  785: 
  786:             self._status = RuntimeBootStatus.STOPPED
  787:             self._failure_reason = failure_reason
  788: 
  789:             self._record_event(
  790:                 stage="shutdown",
  791:                 outcome=(
  792:                     "completed_with_failures"
  793:                     if failure_reason
  794:                     else "completed"
  795:                 ),
  796:                 detail=(
  797:                     failure_reason
  798:                     or context.boot_id
  799:                 ),
  800:             )

  802:     async def fail_shutdown(
  803:         self,
  804:         *,
  805:         context: RuntimeBootContext,
  806:         reason: str,
  807:     ) -> None:
  808:         """Publish an incomplete or failed shutdown."""
  809: 
  810:         normalized_reason = str(reason or "").strip()
  811: 
  812:         if not normalized_reason:
  813:             raise RuntimeBootConfigurationError(
  814:                 "A shutdown-failure reason is required."
  815:             )
  816: 
  817:         async with self._lock:

  393:     async def shutdown(
  394:         self,
  395:     ) -> RuntimeShutdownContext:
  396:         async with self._lock:
  397:             if (
  398:                 self._status
  399:                 is RuntimeShutdownStatus.STOPPED
  400:                 and self._context is not None
  401:             ):
  402:                 self._record_event(
  403:                     stage="shutdown",
  404:                     outcome="already_stopped",
  405:                     detail=self._context.shutdown_id,
  406:                 )
  407: 
  408:                 return self._context
  409: 
  410:             boot_context = (
  411:                 self._boot_process.current_context
  412:             )
  413: 
  414:             if boot_context is None:
  415:                 raise RuntimeShutdownConfigurationError(
  416:                     "No active runtime boot context is available."
  417:                 )
  418: 
  419:             self._attempt += 1
  420:             self._shutdown_id = (
  421:                 f"runtime_shutdown_{self._attempt}_"
  422:                 f"{boot_context.boot_id}"
  423:             )
  424:             self._status = (
  425:                 RuntimeShutdownStatus.SHUTTING_DOWN
  426:             )
  427:             self._failure_reason = None
  428:             self._context = None
  429: 
  430:             started_at = _utc_now()
  431: 
  432:             self._record_event(
  433:                 stage="shutdown",
  434:                 outcome="started",
  435:                 detail=boot_context.boot_id,
  436:             )
  437: 
  438:             failures: list[str] = []
  439:             timeout_seconds = self._resolve_timeout(
  440:                 boot_context
  441:             )
  442: 
  443:             try:
  444:                 await self._boot_process.begin_shutdown(
  445:                     context=boot_context
  446:                 )
  447: 
  448:                 drain_performed = await asyncio.wait_for(
  449:                     self._perform_shutdown(
  450:                         context=boot_context,
  451:                         failures=failures,
  452:                     ),
  453:                     timeout=timeout_seconds,
  454:                 )
  455: 
  456:                 completed_at = _utc_now()
  457: 
  458:                 failure_reason = (
  459:                     "; ".join(failures)
  460:                     if failures
  461:                     else None
  462:                 )
  463: 
  464:                 await self._boot_process.complete_shutdown(
  465:                     context=boot_context,
  466:                     failure_reason=failure_reason,
  467:                 )
  468: 
  469:                 shutdown_context = RuntimeShutdownContext(
  470:                     shutdown_id=self._shutdown_id,
  471:                     attempt=self._attempt,
  472:                     boot_id=boot_context.boot_id,
  473:                     started_at=started_at,
  474:                     completed_at=completed_at,
  475:                     drain_requested=(
  476:                         self._drain_before_stop
  477:                     ),
  478:                     drain_performed=drain_performed,
  479:                     service_count=(
  480:                         boot_context
  481:                         .service_registry
  482:                         .service_count
  483:                     ),
  484:                     failures=tuple(failures),
  485:                     kernel_state=(
  486:                         boot_context.kernel.state.value
  487:                     ),
  488:                     lifecycle_phase=(
  489:                         boot_context
  490:                         .lifecycle_manager
  491:                         .phase
  492:                         .value
  493:                     ),
  494:                     boot_status=(
  495:                         self._boot_process
  496:                         .status
  497:                         .value
  498:                     ),
  499:                 )
  500: 
  501:                 self._context = shutdown_context
  502:                 self._status = (
  503:                     RuntimeShutdownStatus.STOPPED
  504:                 )
  505:                 self._failure_reason = failure_reason
  506: 
  507:                 self._record_event(
  508:                     stage="shutdown",
  509:                     outcome=(
  510:                         "completed_with_failures"
  511:                         if failures
  512:                         else "completed"
  513:                     ),
  514:                     detail=failure_reason,
  515:                 )
  516: 
  517:                 if (
  518:                     failures
  519:                     and self._raise_on_failure
  520:                 ):
  521:                     raise RuntimeShutdownFailureError(
  522:                         shutdown_context
  523:                     )
  524: 
  525:                 return shutdown_context
  526: 
  527:             except RuntimeShutdownFailureError:
  528:                 raise
  529: 
  530:             except asyncio.TimeoutError as exc:
  531:                 self._status = (
  532:                     RuntimeShutdownStatus.FAILED
  533:                 )
  534:                 self._failure_reason = (
  535:                     "Runtime shutdown exceeded "
  536:                     f"{timeout_seconds} seconds."
  537:                 )
  538: 
  539:                 await self._boot_process.fail_shutdown(
  540:                     context=boot_context,
  541:                     reason=self._failure_reason,
  542:                 )
  543: 
  544:                 self._record_event(
  545:                     stage="shutdown",
  546:                     outcome="timeout",
  547:                     detail=self._failure_reason,
  548:                 )
  549: 
  550:                 raise RuntimeShutdownTimeoutError(
  551:                     timeout_seconds=timeout_seconds
  552:                 ) from exc
  553: 
  554:             except Exception as exc:
  555:                 self._status = (
  556:                     RuntimeShutdownStatus.FAILED
  557:                 )
  558:                 self._failure_reason = (
  559:                     f"{type(exc).__name__}: {exc}"

  396:     async def _execute_hook(
  397:         self,

  399:         service_key: str,
  400:         service: object,
  401:         action: str,
  402:         timeout_seconds: float,

  404:         resolved = self._resolve_hook(
  405:             service,
  406:             action,
  407:         )
  408: 
  409:         if resolved is None:
  410:             self._record_event(
  411:                 action=action,
  412:                 status="skipped",
  413:                 service_key=service_key,
  414:                 detail="No lifecycle hook was defined.",
  415:             )
  416:             return None
  417: 
  418:         hook_name, hook = resolved
  419: 
  420:         async def invoke() -> Any:
  421:             result = hook()

  423:             if inspect.isawaitable(result):
  424:                 return await result
  425: 
  426:             return result
  427: 
  428:         self._record_event(
  429:             action=action,
  430:             status="started",
  431:             service_key=service_key,
  432:             detail=f"hook={hook_name}",
  433:         )

  435:         try:
  436:             result = await asyncio.wait_for(
  437:                 invoke(),
  438:                 timeout=timeout_seconds,
  439:             )

  442:                 f"Hook {hook_name!r} exceeded "
  443:                 f"{timeout_seconds} seconds."

  448:                 action=action,

  450:             ) from exc
  451:         except RuntimeLifecycleServiceError:
  452:             raise
  453:         except Exception as exc:
  454:             raise RuntimeLifecycleServiceError(
  455:                 service_key=service_key,
  456:                 action=action,
  457:                 cause=exc,
  458:             ) from exc

  465:                     "The service reported not ready."
  466:                 ),
  467:             )
  468: 
  469:         self._record_event(
  470:             action=action,
  471:             status="completed",
  472:             service_key=service_key,
  473:             detail=f"hook={hook_name}",
  474:         )
  475: 
  476:         return result

  507:                     status="failed",

  516:                     and not continue_after_critical_failure

  296:     def fingerprint(self) -> str:
  297:         """
  298:         Return a deterministic value fingerprint.
  299: 
  300:         Source labels are intentionally excluded. Equal effective
  301:         configuration values produce equal fingerprints regardless of
  302:         whether they came from defaults, environment variables, or
  303:         explicit overrides.
  304:         """
  305: 
  306:         payload = json.dumps(
  307:             self.to_dict(
  308:                 include_sources=False,
  309:                 redact_sensitive=False,
  310:             ),
  311:             sort_keys=True,
  312:             separators=(",", ":"),
  313:             ensure_ascii=False,
  314:         ).encode("utf-8")

  316:         return hashlib.sha256(payload).hexdigest()

  318:     def _validate(self) -> None:
  319:         if not _CONFIGURATION_VERSION_PATTERN.fullmatch(
  320:             self.configuration_schema_version
  321:         ):
  322:             raise RuntimeConfigurationValidationError(
  323:                 "configuration_schema_version must use "
  324:                 "MAJOR.MINOR.PATCH format."
  325:             )
  326: 
  327:         if (
  328:             self.configuration_schema_version
  329:             != CONFIGURATION_SCHEMA_VERSION
  330:         ):
  331:             raise RuntimeConfigurationValidationError(
  332:                 "Unsupported configuration schema version: "
  333:                 f"{self.configuration_schema_version!r}."
  334:             )
  335: 
  336:         if not _RUNTIME_ID_PATTERN.fullmatch(
  337:             self.runtime_id
  338:         ):
  339:             raise RuntimeConfigurationValidationError(
  340:                 "runtime_id must be 3-128 characters and "
  341:                 "contain only letters, numbers, dots, "
  342:                 "underscores, colons, or hyphens."
  343:             )
  344: 
  345:         if not self.product_name.strip():

  347:                 "product_name must not be empty."
  348:             )
  349:

  351:             raise RuntimeConfigurationValidationError(
  352:                 "product_name must not exceed 128 characters."

  356:             self.environment
  357:             not in ALLOWED_ENVIRONMENT_NAMES

  359:             raise RuntimeConfigurationValidationError(
  360:                 "environment must be one of: "
  361:                 + ", ".join(

  364:             )
  365: 
  366:         if self.log_level not in _ALLOWED_LOG_LEVELS:

  368:                 "log_level must be one of: "

  370:             )

  377:         if self.state_backend not in _ALLOWED_STATE_BACKENDS:
  378:             raise RuntimeConfigurationValidationError(
  379:                 "state_backend must be one of: "
  380:                 + ", ".join(
  381:                     sorted(_ALLOWED_STATE_BACKENDS)
  382:                 )
  383:             )
  384: 
  385:         if not (
  386:             0.1
  387:             <= self.startup_timeout_seconds
  388:             <= 3600.0
  389:         ):
  390:             raise RuntimeConfigurationValidationError(
  391:                 "startup_timeout_seconds must be between "
  392:                 "0.1 and 3600 seconds."
  393:             )
  394: 
  395:         if not (
  396:             0.1
  397:             <= self.shutdown_timeout_seconds
  398:             <= 3600.0
  399:         ):
  400:             raise RuntimeConfigurationValidationError(
  401:                 "shutdown_timeout_seconds must be between "
  402:                 "0.1 and 3600 seconds."
  403:             )
  404: 
  405:         if not (
  406:             0.05
  407:             <= self.queue_poll_interval_seconds
  408:             <= 300.0
  409:         ):
  410:             raise RuntimeConfigurationValidationError(
  411:                 "queue_poll_interval_seconds must be between "
  412:                 "0.05 and 300 seconds."
  413:             )
  414: 
  415:         if not 1 <= self.max_concurrency <= 1024:
  416:             raise RuntimeConfigurationValidationError(
  417:                 "max_concurrency must be between 1 and 1024."
  418:             )
  419: 
  420:         if self.is_production and self.debug:
  421:             raise RuntimeConfigurationValidationError(
  422:                 "debug must be disabled in production."
  423:             )
  424: 
  425:         if (
  426:             self.is_production

  443:         missing_sources = sorted(
  444:             field
  445:             for field in _CONFIGURATION_FIELDS

  450:             raise RuntimeConfigurationValidationError(
  451:                 "Missing configuration-source records: "
  452:                 + ", ".join(missing_sources)

  456: class RuntimeConfigurationLoader:
  457:     """
  458:     Resolve runtime configuration from controlled sources.
  459: 
  460:     Precedence, from lowest to highest:
  461: 
  462:     1. Built-in defaults
  463:     2. Runtime-prefixed environment variables
  464:     3. Explicit overrides
  465: 
  466:     The loader does not mutate os.environ and does not create any
  467:     configured directories.
  468:     """
  469: 
  470:     __slots__ = (
  471:         "_project_root",
  472:         "_environment_prefix",
  473:     )
  474: 
  475:     def __init__(
  476:         self,
  477:         *,
  478:         project_root: Path,
  479:         environment_prefix: str = (
  480:             CONFIGURATION_ENV_PREFIX
  481:         ),
  482:     ) -> None:
  483:         resolved_root = Path(project_root).expanduser()
  484: 
  485:         if not resolved_root.is_absolute():
  486:             resolved_root = resolved_root.resolve()
  487: 
  488:         normalized_prefix = str(
  489:             environment_prefix or ""
  490:         ).strip().upper()
  491: 
  492:         if not normalized_prefix:
  493:             raise RuntimeConfigurationSourceError(
  494:                 "environment_prefix must not be empty."
  495:             )
  496: 
  497:         if not normalized_prefix.endswith("_"):
  498:             raise RuntimeConfigurationSourceError(
  499:                 "environment_prefix must end with an underscore."
  500:             )
  501: 
  502:         self._project_root = resolved_root.resolve()
  503:         self._environment_prefix = normalized_prefix
  504: 
  505:     @property
  506:     def project_root(self) -> Path:
  507:         return self._project_root
  508: 
  509:     @property
  510:     def environment_prefix(self) -> str:
  511:         return self._environment_prefix
  512: 
  513:     def defaults(self) -> Mapping[str, Any]:
  514:         return MappingProxyType(dict(_DEFAULT_VALUES))
  515: 
  516:     def load(
  517:         self,
  518:         *,
  519:         environ: Mapping[str, str] | None = None,
  520:         overrides: Mapping[str, Any] | None = None,
  521:     ) -> RuntimeConfiguration:
  522:         environment_values = (
  523:             os.environ
  524:             if environ is None
  525:             else environ
  526:         )
  527: 
  528:         explicit_overrides = (
  529:             {}
  530:             if overrides is None
  531:             else dict(overrides)
  532:         )
  533: 
  534:         unknown_override_fields = sorted(
  535:             set(explicit_overrides)
  536:             - _CONFIGURATION_FIELDS
  537:         )
  538: 
  539:         if unknown_override_fields:
  540:             raise RuntimeConfigurationSourceError(
  541:                 "Unknown runtime configuration overrides: "
  542:                 + ", ".join(unknown_override_fields)
  543:             )
  544: 
  545:         if "field_sources" in explicit_overrides:
  546:             raise RuntimeConfigurationSourceError(
  547:                 "field_sources cannot be supplied as an override."
  548:             )
  549: 
  550:         values = dict(_DEFAULT_VALUES)
  551: 
  552:         sources = {
  553:             field: "default"
  554:             for field in _CONFIGURATION_FIELDS
  555:         }
  556: 
  557:         environment_updates = (
  558:             self._collect_environment_updates(
  559:                 environment_values
  560:             )
  561:         )
  562: 
  563:         for field, update in environment_updates.items():
  564:             raw_value, source_name = update
  565:             values[field] = self._coerce_value(
  566:                 field,
  567:                 raw_value,
  568:             )
  569:             sources[field] = source_name
  570: 
  571:         for field, raw_value in explicit_overrides.items():
  572:             values[field] = self._coerce_value(
  573:                 field,
  574:                 raw_value,
  575:             )
  576:             sources[field] = "override"
  577: 
  578:         values["data_root"] = self._resolve_data_root(
  579:             values["data_root"]
  580:         )
  581: 
  582:         frozen_sources = MappingProxyType(
  583:             dict(sources)
  584:         )
  585: 
  586:         return RuntimeConfiguration(
  587:             configuration_schema_version=str(
  588:                 values[
  589:                     "configuration_schema_version"
  590:                 ]
  591:             ),
  592:             runtime_id=str(values["runtime_id"]).strip(),
  593:             product_name=str(
  594:                 values["product_name"]
  595:             ).strip(),
  596:             environment=str(
  597:                 values["environment"]
  598:             ).strip().lower(),
  599:             debug=bool(values["debug"]),
  600:             log_level=str(
  601:                 values["log_level"]
  602:             ).strip().upper(),
  603:             data_root=values["data_root"],
  604:             state_backend=str(
  605:                 values["state_backend"]
  606:             ).strip().lower(),
  607:             worker_enabled=bool(
  608:                 values["worker_enabled"]
  609:             ),
  610:             startup_timeout_seconds=float(
  611:                 values["startup_timeout_seconds"]
  612:             ),
  613:             shutdown_timeout_seconds=float(
  614:                 values["shutdown_timeout_seconds"]
  615:             ),
  616:             queue_poll_interval_seconds=float(
  617:                 values[
  618:                     "queue_poll_interval_seconds"
  619:                 ]
  620:             ),
  621:             max_concurrency=int(
  622:                 values["max_concurrency"]
  623:             ),
  624:             strict_compatibility=bool(
  625:                 values["strict_compatibility"]
  626:             ),
  627:             schema_auto_migrate=bool(
  628:                 values["schema_auto_migrate"]
  629:             ),
  630:             field_sources=frozen_sources,
  631:         )

  633:     def _collect_environment_updates(
  634:         self,
  635:         environ: Mapping[str, str],
  636:     ) -> dict[str, tuple[str, str]]:
  637:         updates: dict[str, tuple[str, str]] = {}
  638:         unknown_keys: list[str] = []
  639: 
  640:         for raw_name, raw_value in environ.items():
  641:             name = str(raw_name).strip()
  642:             upper_name = name.upper()
  643: 
  644:             if not upper_name.startswith(
  645:                 self._environment_prefix
  646:             ):
  647:                 continue
  648: 
  649:             suffix = upper_name[
  650:                 len(self._environment_prefix):
  651:             ]
  652: 
  653:             field = _ENVIRONMENT_FIELD_MAP.get(
  654:                 suffix
  655:             )
  656: 
  657:             if field is None:
  658:                 unknown_keys.append(name)
  659:                 continue
  660: 
  661:             updates[field] = (
  662:                 str(raw_value),
  663:                 f"environment:{name}",
  664:             )
  665: 
  666:         if unknown_keys:
  667:             raise RuntimeConfigurationSourceError(
  668:                 "Unknown runtime environment variables: "
  669:                 + ", ".join(sorted(unknown_keys))
  670:             )
  671: 
  672:         return updates

  674:     def _coerce_value(
  675:         self,
  676:         field: str,
  677:         raw_value: Any,
  678:     ) -> Any:
  679:         if field not in _CONFIGURATION_FIELDS:
  680:             raise RuntimeConfigurationSourceError(
  681:                 f"Unknown runtime configuration field: "
  682:                 f"{field!r}"
  683:             )
  684: 
  685:         if field in {
  686:             "debug",
  687:             "worker_enabled",
  688:             "strict_compatibility",
  689:             "schema_auto_migrate",
  690:         }:
  691:             return self._parse_boolean(
  692:                 field,
  693:                 raw_value,
  694:             )
  695: 
  696:         if field == "max_concurrency":
  697:             return self._parse_integer(
  698:                 field,
  699:                 raw_value,
  700:             )
  701: 
  702:         if field in {
  703:             "startup_timeout_seconds",
  704:             "shutdown_timeout_seconds",
  705:             "queue_poll_interval_seconds",
  706:         }:
  707:             return self._parse_float(
  708:                 field,
  709:                 raw_value,
  710:             )
  711: 
  712:         if field == "data_root":
  713:             if isinstance(raw_value, Path):
  714:                 return raw_value
  715: 
  716:             return Path(
  717:                 str(raw_value or "").strip()
  718:             )
  719: 
  720:         return str(raw_value or "").strip()
  721: 
  722:     def _parse_boolean(
  723:         self,
  724:         field: str,
  725:         raw_value: Any,
  726:     ) -> bool:
  727:         if isinstance(raw_value, bool):
  728:             return raw_value
  729: 
  730:         normalized = str(raw_value or "").strip().lower()
  731: 
  732:         if normalized in _TRUE_VALUES:
  733:             return True
  734: 
  735:         if normalized in _FALSE_VALUES:
  736:             return False
  737: 
  738:         raise RuntimeConfigurationValidationError(
  739:             f"{field} must be a recognized boolean value."
  740:         )
  741: 
  742:     def _parse_integer(
  743:         self,
  744:         field: str,
  745:         raw_value: Any,
  746:     ) -> int:
  747:         if isinstance(raw_value, bool):
  748:             raise RuntimeConfigurationValidationError(
  749:                 f"{field} must be an integer, not a boolean."
  750:             )
  751: 
  752:         try:
  753:             return int(str(raw_value).strip())
  754:         except (TypeError, ValueError) as exc:
  755:             raise RuntimeConfigurationValidationError(
  756:                 f"{field} must be a valid integer."
  757:             ) from exc
  758: 
  759:     def _parse_float(
  760:         self,
  761:         field: str,
  762:         raw_value: Any,
  763:     ) -> float:
  764:         if isinstance(raw_value, bool):
  765:             raise RuntimeConfigurationValidationError(
  766:                 f"{field} must be numeric, not a boolean."
  767:             )
  768: 
  769:         try:
  770:             return float(str(raw_value).strip())
  771:         except (TypeError, ValueError) as exc:
  772:             raise RuntimeConfigurationValidationError(
  773:                 f"{field} must be a valid number."
  774:             ) from exc
  775:
